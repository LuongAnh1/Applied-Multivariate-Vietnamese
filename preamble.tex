\newcommand{\BookChapterTitle}[1]{%
  \vspace*{-1.0em}%
  \noindent\begin{tikzpicture}
    \path[use as bounding box] (0,-0.46in) rectangle (\textwidth,1.38in);
    \begin{scope}[overlay]
      \fill[BookGray] (-0.55in,0) rectangle (\textwidth,1.02in);
      \foreach \x in {-0.45,-0.20,...,4.6} {
        \draw[BookGrayDark!32, line width=0.3pt] (\x,0.08) -- ++(1.1,0.86);
      }
      \node[anchor=west, inner sep=0pt] at (0.30in,1.22in)
        {\fontsize{17}{19}\selectfont\itshape\addfontfeatures{LetterSpace=10}Chapter};
      \node[anchor=west, inner sep=0pt] at (0.58in,0.28in)
        {\fontsize{84}{84}\selectfont\thechapter};
      \node[anchor=west, text width=\textwidth, inner sep=0pt] at (0,-0.34in)
        {\fontsize{22}{26}\selectfont\scshape\MakeLowercase{#1}};
    \end{scope}
  \end{tikzpicture}%
  \par\vspace{0.8em}%
}
