\begin{frame}{Bài tập và ví dụ tính toán}
  \small
  {\large\textbf{Bài 5.1: }\par}
  \vspace{0.15cm}
  \begin{block}{Đề bài}
    Cho ma trận dữ liệu
    \[
      \mat{X}=
      \begin{bmatrix}
        2 & 12\\
        8 & 9\\
        6 & 9\\
        8 & 10
      \end{bmatrix}.
    \]
    \begin{enumerate}
      \item[(a)] Tính giá trị của thống kê \(T^2\) để kiểm định \(H_0:\vect{\mu}\trans=[7,\ 11]\).
      \item[(b)] Xác định phân phối của \(T^2\) trong trường hợp ở câu (a).
      \item[(c)] Sử dụng kết quả ở câu (a) và (b), hãy kiểm định \(H_0\) ở mức ý nghĩa \(\alpha=0{,}05\). Nêu kết luận.
    \end{enumerate}
  \end{block}
\end{frame}
 
\begin{frame}{Bài tập và ví dụ tính toán}
  \footnotesize
  {\large\textbf{Bài 5.1(a): Trung bình mẫu, \(\mat{S}\) và \(\mat{S}\inv\)}\par}
  \[
    \bar{\vect{x}}=
    \begin{bmatrix}\bar{x}_1\\ \bar{x}_2\end{bmatrix}=
    \begin{bmatrix}\dfrac{2+8+6+8}{4}\\[8pt] \dfrac{12+9+9+10}{4}\end{bmatrix}=
    \begin{bmatrix}6\\ 10\end{bmatrix}
  \]
  và
  \[
    s_{11}=\frac{(2-6)^2+(8-6)^2+(6-6)^2+(8-6)^2}{3}=8,
  \]
  \[
    s_{12}=\frac{(2-6)(12-10)+(8-6)(9-10)+(6-6)(9-10)+(8-6)(10-10)}{3}=\frac{-10}{3},
  \]
  \[
    s_{22}=\frac{(12-10)^2+(9-10)^2+(9-10)^2+(10-10)^2}{3}=2.
  \]
  Do đó,
  \[
    \mat{S}=
    \begin{bmatrix}
      8 & -\dfrac{10}{3}\\[8pt]
      -\dfrac{10}{3} & 2
    \end{bmatrix},
    \qquad
    \mat{S}\inv=\frac{9}{44}
    \begin{bmatrix}
      2 & \dfrac{10}{3}\\[8pt]
      \dfrac{10}{3} & 8
    \end{bmatrix}.
  \]
\end{frame}
 
 
 
\begin{frame}{Bài tập và ví dụ tính toán}
  \small
  {\large\textbf{Bài 5.1(a): Giá trị \(T^2\)}\par}
  \vspace{0.05cm}
  Đặt \(\vect{d}=\bar{\vect{x}}-\vect{\mu}_0=[\,6-7,\ 10-11\,]\trans=[-1,\ -1]\trans\). Khi đó, với \(n=4\):
  \[
  \begin{aligned}
    T^2 &= 4\,\vect{d}\trans\mat{S}\inv\vect{d}\\
    &= 4\,[\,-1,\ -1\,]\,\frac{9}{44}
      \begin{bmatrix}2&\dfrac{10}{3}\\[8pt]\dfrac{10}{3}&8\end{bmatrix}
      \begin{bmatrix}-1\\-1\end{bmatrix}\\
    &= 4\,[\,-1,\ -1\,]
      \begin{bmatrix}-\dfrac{12}{11}\\[8pt]-\dfrac{51}{22}\end{bmatrix}\\
    &= 4\cdot\frac{75}{22}=\frac{150}{11}\approx 13{,}64.
  \end{aligned}
  \]
\end{frame}
 
\begin{frame}{Bài tập và ví dụ tính toán}
   \small
  {\large\textbf{Bài 5.1(b), (c): Phân phối của \(T^2\) và kết luận}\par}
  \vspace{-0.05cm}
  \begin{block}{(b) Phân phối của \(T^2\)}
    Giả thiết các hàng của \(\mat{X}\) là mẫu ngẫu nhiên từ tổng thể chuẩn hai chiều \(N_2(\vect{\mu},\boldsymbol{\Sigma})\). Trước khi mẫu được chọn, \(T^2\) có phân phối của biến ngẫu nhiên
    \[
      \frac{(n-1)p}{n-p}F_{p,n-p}=\frac{(4-1)2}{(4-2)}F_{2,4-2}=3F_{2,2}.
    \]
  \end{block}

  \begin{block}{(c) Kiểm định \(H_0:\vect{\mu}\trans=[7,\ 11]\) ở mức \(\alpha=0{,}05\)}
    So sánh giá trị quan sát \(T^2=13{,}64\) với giá trị tới hạn
    \[
      \frac{(n-1)p}{n-p}F_{p,n-p}(0{,}05)=\frac{(4-1)2}{(4-2)}F_{2,2}(0{,}05)=3.19=57.
    \]
  \end{block}
 
\end{frame}

\begin{frame}{Bài tập và ví dụ tính toán}

\begin{frame}{Bài tập và ví dụ tính toán}
  \footnotesize
  {\large\textbf{Bài 5.2: Trung bình mới, \(\mat{S}_y\) và \(\mat{S}_y\inv\)}\par}
  \vspace{0.05cm}
  Từ \(\mat{Y}\) (\(n=3\)), ta có
  \[
    \bar{\vect{y}}=
    \begin{bmatrix}
      \dfrac{-3+4+5}{3}\\[8pt]
      \dfrac{15+16+11}{3}
    \end{bmatrix}=
    \begin{bmatrix}
      2\\
      14
    \end{bmatrix}=\mat{C}\bar{\vect{x}}
  \]
  và
  \[
    s_{y,11}=\frac{(-3-2)^2+(4-2)^2+(5-2)^2}{2}=19,
  \]
  \[
    s_{y,12}=\frac{(-3-2)(15-14)+(4-2)(16-14)+(5-2)(11-14)}{2}=-5,
  \]
  \[
    s_{y,22}=\frac{(15-14)^2+(16-14)^2+(11-14)^2}{2}=7.
  \]
  Do đó, với \(\det\mat{S}_y=19\cdot7-(-5)^2=108\),
  \[
    \mat{S}_y=
    \begin{bmatrix}
      19 & -5\\
      -5 & 7
    \end{bmatrix},
    \qquad
    \mat{S}_y\inv=\frac{1}{108}
    \begin{bmatrix}
      7 & 5\\
      5 & 19
    \end{bmatrix}.
  \]
\end{frame}
 
 
 
\begin{frame}{Bài tập và ví dụ tính toán}
  \small
  {\large\textbf{Bài 5.2: \(T^2\) của dữ liệu mới}\par}
  \vspace{0.05cm}
  Với \(n=3\) và giả thuyết mới \(\mat{C}\vect{\mu}_0=[4,\ 14]\trans\):
  \[
  \begin{aligned}
    T^2_y &= 3\,[\,2-4,\ 14-14\,]\,\frac{1}{108}
      \begin{bmatrix}7&5\\5&19\end{bmatrix}
      \begin{bmatrix}2-4\\14-14\end{bmatrix}\\
    &= 3\,[\,-2,\ 0\,]\,\frac{1}{108}
      \begin{bmatrix}7&5\\5&19\end{bmatrix}
      \begin{bmatrix}-2\\0\end{bmatrix}
      =\frac{3\cdot 4\cdot 7}{108}=\frac{7}{9}.
  \end{aligned}
  \]
 
  \begin{alertblock}{Kết luận}
    \(T^2_y=\dfrac{7}{9}\) bằng đúng \(T^2\) của dữ liệu gốc, nên \(T^2\) không thay đổi. 
  \end{alertblock}
\end{frame}

\begin{frame}{Bài tập và ví dụ tính toán}
  \small
  {\large\textbf{Bài 5.3:}\par}
  \vspace{0.1cm}
  \begin{block}{Đề bài}
    \begin{enumerate}
      \item[(a)] Sử dụng biểu thức (5-15) để tính \(T^2\) cho dữ liệu trong Bài tập 5.1.
      \item[(b)] Sử dụng dữ liệu trong Bài tập 5.1 để tính tỉ số hợp lý \(\Lambda\) trong (5-13). Đồng thời tính lambda Wilks \(\Lambda^{\frac{2}{n}}\).
    \end{enumerate}
  \end{block}
  \vspace{-0.1cm}
  \footnotesize
  \begin{block}{Biểu thức (5-15) và (5-13)}
    \vspace{-0.2cm}
    \[
      T^2=\frac{(n-1)\left|\sum_{j=1}^n(\vect{x}_j-\vect{\mu}_0)(\vect{x}_j-\vect{\mu}_0)\trans\right|}
      {\left|\sum_{j=1}^n(\vect{x}_j-\bar{\vect{x}})(\vect{x}_j-\bar{\vect{x}})\trans\right|}-(n-1)
    \]
    \[
      \Lambda=\left(\frac{|\hat{\boldsymbol{\Sigma}}|}{|\hat{\boldsymbol{\Sigma}}_0|}\right)^{\frac{n}{2}}
      =\left(\frac{\left|\sum_{j=1}^n(\vect{x}_j-\bar{\vect{x}})(\vect{x}_j-\bar{\vect{x}})\trans\right|}
      {\left|\sum_{j=1}^n(\vect{x}_j-\vect{\mu}_0)(\vect{x}_j-\vect{\mu}_0)\trans\right|}\right)^{\frac{n}{2}}<c_\alpha
    \]
  \end{block}
\end{frame}
 
\begin{frame}{Bài tập và ví dụ tính toán}
  {\large\textbf{Bài 5.3: Ma trận \(\mat{A}\) (độ lệch so với \(\bar{\vect{x}}\))}\par}
  \vspace{0.05cm}
  Dùng dữ liệu Bài 5.1 (\(n=4\), \(\bar{\vect{x}}=[6,\ 10]\trans\)). \\
  Đặt
  \(\mat{A}=\sum_{j=1}^n(\vect{x}_j-\bar{\vect{x}})(\vect{x}_j-\bar{\vect{x}})\trans\) 
  \[
    A_{11}=(2-6)^2+(8-6)^2+(6-6)^2+(8-6)^2=24,
  \]
  \[
    A_{12}=(2-6)(12-10)+(8-6)(9-10)+(6-6)(9-10)+(8-6)(10-10)=-10,
  \]
  \[
    A_{22}=(12-10)^2+(9-10)^2+(9-10)^2+(10-10)^2=6.
  \]
  Do đó
  \[
  \begin{aligned}
    &\mat{A}=
    \begin{bmatrix}
      24 & -10\\
      -10 & 6
    \end{bmatrix}=3\mat{S},\\
    &|\mat{A}|=24\cdot 6-(-10)^2=44.
    \end{aligned}
  \]
\end{frame}
 
\begin{frame}{Bài tập và ví dụ tính toán}
  {\large\textbf{Bài 5.3: Ma trận \(\mat{B}\) (độ lệch so với \(\vect{\mu}_0\))}\par}
  \vspace{0.05cm}
  Đặt \(\mat{B}=\sum_{j=1}^n(\vect{x}_j-\vect{\mu}_0)(\vect{x}_j-\vect{\mu}_0)\trans\) với \(\vect{\mu}_0=[7,\ 11]\trans\):
  \[
    B_{11}=(2-7)^2+(8-7)^2+(6-7)^2+(8-7)^2=28,
  \]
  \[
    B_{12}=(2-7)(12-11)+(8-7)(9-11)+(6-7)(9-11)+(8-7)(10-11)=-6,
  \]
  \[
    B_{22}=(12-11)^2+(9-11)^2+(9-11)^2+(10-11)^2=10.
  \]
  Do đó
  \[
  \begin{aligned}
    &\mat{B}=
    \begin{bmatrix}
      28 & -6\\
      -6 & 10
    \end{bmatrix},\\
    &|\mat{B}|=28\cdot 10-(-6)^2=244.
    \end{aligned}
  \]
\end{frame}
 
\begin{frame}{Bài tập và ví dụ tính toán}
  \small
  {\large\textbf{Bài 5.3: \(T^2\), tỉ số hợp lý và lambda Wilks}\par}
  \vspace{0.1cm}
  Vì \(\hat{\boldsymbol{\Sigma}}=\dfrac{\mat{A}}{n}\) và \(\hat{\boldsymbol{\Sigma}}_0=\dfrac{\mat{B}}{n}\), hệ số \(\dfrac{1}{n}\) triệt tiêu trong tỉ số:
  \(\dfrac{|\hat{\boldsymbol{\Sigma}}_0|}{|\hat{\boldsymbol{\Sigma}}|}=\dfrac{|\mat{B}|}{|\mat{A}|}=\dfrac{244}{44}=\dfrac{61}{11}\).
 
    \begin{block}{(a) \(T^2\) theo (5-15)}
      \[
        T^2=(n-1)\frac{|\mat{B}|}{|\mat{A}|}-(n-1)
        =3\cdot\frac{61}{11}-3=\frac{150}{11}\approx13{,}64
      \]
    \end{block}
    \vspace{-0.1cm}
    \begin{block}{(b) \(\Lambda\) và lambda Wilks}
      \[
        \Lambda^{\frac{2}{n}}=\frac{|\mat{A}|}{|\mat{B}|}=\frac{11}{61}\approx0{,}180
      \]
      \[
        \Lambda=\left(\frac{11}{61}\right)^{\frac{4}{2}}=\frac{121}{3721}\approx0{,}0325
      \]
    \end{block}
  \end{frame}

\begin{frame}{Bài tập và ví dụ tính toán}
  \small
  {\large\textbf{Bài 5.4(a): trị riêng và hướng trục}\par}
  \vspace{0.1cm}
  Trục của ellipsoid đi theo các vectơ riêng chuẩn hóa của \(\mat{S}\). Trị riêng là nghiệm của
  \[
    |\mat{S}-\lambda\mat{I}|=0
    \quad\Longrightarrow\quad
    \lambda^3-206{,}295\lambda^2+1175{,}180\lambda-1181{,}529=0.
  \]

  \begin{block}{Cách chọn vectơ riêng}
    Với mỗi \(\lambda_i\), giải hệ thuần nhất
    \((\mat{S}-\lambda_i\mat{I})\vect{v}_i=\vect{0}\).
    Vectơ riêng chỉ xác định đến hằng số nhân khác \(0\), nên có thể đặt một thành phần thuận tiện bằng \(1\), rồi chuẩn hóa
    \(\vect{e}_i=\vect{v}_i/\|\vect{v}_i\|\).
  \end{block}

\begin{frame}{Bài tập và ví dụ tính toán}
  \small
  {\large\textbf{Bài 5.4(b): cách dựng Q--Q plot}\par}
  \vspace{0.1cm}
  Theo Mục 4.6, trang 178--179, Q--Q plot so sánh phân vị mẫu với phân vị lý thuyết của phân phối chuẩn.
  Với mỗi biến \(X_k\), thực hiện:
  \begin{enumerate}
    \item Sắp xếp mẫu: \(x_{(1)k}\le x_{(2)k}\le \cdots \le x_{(20)k}\).
    \item Gán vị trí xác suất cho quan sát thứ \(j\): \(p_j=(j-\frac{1}{2})/n=(j-\frac{1}{2})/20\).
    \item Tính phân vị chuẩn lý thuyết: \(q_j=\Phi^{-1}(p_j)\).
  \end{enumerate}
  Các điểm được vẽ là \((q_j,\ x_{(j)k})\). Trục hoành là phân vị chuẩn lý thuyết, trục tung là dữ liệu đã sắp xếp.

  \vspace{0.15cm}
  {\footnotesize\textit{Chú thích:} Chuẩn đa biến là giả thiết về cả vectơ \(X=(X_1,X_2,X_3)^T\). Q--Q plot kiểm tra từng biến biên, còn scatter plot giúp nhìn quan hệ theo từng cặp biến.\par}
\end{frame}

\begin{frame}{Bài tập và ví dụ tính toán}
  \small
  {\large\textbf{Bài 5.5: quyết định kiểm định}\par}
  \vspace{0.1cm}
  Kiểm định
  \[
    H_0:\vect{\mu}\trans=[0{,}55,\ 0{,}60]
    \qquad\text{với}\qquad
    \alpha=0{,}05.
  \]
  Giá trị tới hạn:
  \[
    \frac{(n-1)p}{n-p}F_{p,n-p}(\alpha)
    =
    \frac{41(2)}{40}F_{2,40}(0{,}05)
    =6{,}625.
  \]

  \begin{alertblock}{Kết luận}
    Vì \(T^2=1{,}170<6{,}625\), ta không bác bỏ \(H_0\).
    Kết quả nhất quán với elip tin cậy \(95\%\) trong Hình 5.1, vì điểm \([0{,}55,\ 0{,}60]\trans\) nằm trong miền tin cậy.
  \end{alertblock}
\end{frame}
