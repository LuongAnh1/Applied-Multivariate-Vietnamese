\section{Bài tập và ví dụ tính toán}
\sectionframe{Bài tập và ví dụ tính toán}

\begin{frame}{Bài tập và ví dụ tính toán}
  \begin{block}{Phạm vi nội dung}
    Gom các bài tập, ví dụ số, bảng dữ liệu hoặc minh họa tính toán được nhóm chọn để trình bày.
  \end{block}
  \begin{itemize}
    \item Tóm tắt đề bài hoặc dữ liệu cần dùng.
    \item Trình bày các bước tính chính, tránh dồn quá nhiều biến đổi đại số lên một slide.
    \item Chốt kết luận bằng ngôn ngữ thống kê và bối cảnh bài toán.
  \end{itemize}
\end{frame}
