\begin{frame}{Mục đích của biểu đồ kiểm soát}
  \begin{itemize}
    \item Quy trình \textbf{ổn định}: biến động chỉ do \textbf{nguyên nhân phổ biến} (\textit{common causes}), không có nguyên nhân nào chi phối.
    \item Biểu đồ kiểm soát nhận diện \textbf{nguyên nhân đặc biệt} (\textit{special causes}) từ bên ngoài quy trình thông thường.
    \item Nguyên nhân đặc biệt thường cần khắc phục kịp thời, nhưng cũng có thể gợi ý cải tiến quy trình.
    \item Biểu đồ gồm dữ liệu vẽ theo thời gian và các đường ngang gọi là \textbf{giới hạn kiểm soát}.
  \end{itemize}
\end{frame}

\begin{frame}{Biểu đồ \(\bar{X}\) (đơn biến)}
  \begin{block}{Cách xây dựng}
    \begin{enumerate}
      \item Vẽ các quan sát riêng lẻ hoặc trung bình mẫu theo thời gian.
      \item Vẽ đường trung tâm \(\bar{\bar{x}}\): trung bình của mọi quan sát.
      \item Vẽ các giới hạn kiểm soát:
    \end{enumerate}
    \begin{align*}
      \mathrm{UCL}&=\bar{\bar{x}}+3(\text{độ lệch chuẩn})\\
      \mathrm{LCL}&=\bar{\bar{x}}-3(\text{độ lệch chuẩn})
    \end{align*}
  \end{block}
  \small Quan sát đơn lẻ: độ lệch chuẩn mẫu. Trung bình nhóm con cỡ \(m\): độ lệch chuẩn mẫu chia \(\sqrt{m}\). Mức \(\pm3\) độ lệch chuẩn làm xác suất báo động giả rất nhỏ (giả định dữ liệu chuẩn).
\end{frame}

\begin{frame}{Vì sao cần phương pháp đa biến?}
  \begin{itemize}
    \item Với nhiều đặc tính quan trọng, cần phương pháp đa biến để giám sát tính ổn định.
    \item Tính đến \textbf{tương quan} giữa các đặc tính và kiểm soát được xác suất \textbf{tổng thể} báo động sai.
    \item Tương quan cao khiến việc đánh giá tỷ lệ sai số tổng thể của nhiều biểu đồ đơn biến trở nên bất khả thi.
  \end{itemize}
  \begin{block}{Hai loại biểu đồ phổ biến}
    (i) Biểu đồ dạng elip; \quad (ii) Biểu đồ \(T^2\).
  \end{block}
  Hai tình huống cần xử lý khác nhau: \textbf{theo dõi tính ổn định} của một mẫu cho trước, và \textbf{thiết lập vùng kiểm soát} cho quan sát tương lai.
\end{frame}

\begin{frame}{Theo dõi mẫu các quan sát đa biến riêng lẻ}
  Giả định \(\vect{X}_1,\dots,\vect{X}_n\) độc lập, cùng phân phối \(N_p(\vect{\mu},\vect{\Sigma})\). Theo Kết quả 4.8:
  \begin{align*}
    E(\vect{X}_j-\bar{\vect{X}})&=\vect{0}\\
    \operatorname{Cov}(\vect{X}_j-\bar{\vect{X}})&=\frac{n-1}{n}\,\vect{\Sigma}
  \end{align*}
  \begin{itemize}
    \item Mỗi \(\vect{X}_j-\bar{\vect{X}}\) có phân phối chuẩn, nhưng \textbf{không độc lập} với \(\mat{S}\).
    \item Để lập giới hạn kiểm soát, \textbf{xấp xỉ} \((\vect{X}_j-\bar{\vect{X}})\trans\mat{S}\inv(\vect{X}_j-\bar{\vect{X}})\) có phân phối chi bình phương.
  \end{itemize}
\end{frame}

\begin{frame}{Biểu đồ dạng elip}
  \begin{itemize}
    \item Trực quan nhất, nhưng \textbf{chỉ dùng cho hai biến}: vẽ cặp \((x_{j1},x_{j2})\) của đơn vị thứ \(j\).
  \end{itemize}
  \begin{block}{Elip chất lượng 95\%}
    Gồm mọi \(\vect{x}\) thỏa
    \begin{equation}
      (\vect{x}-\bar{\vect{x}})\trans\mat{S}\inv(\vect{x}-\bar{\vect{x}})\le \chi^2_2(.05)
      \tag{5-32}
    \end{equation}
  \end{block}
  Điểm nằm ngoài elip là tín hiệu có thể có nguyên nhân đặc biệt.
\end{frame}

\begin{frame}{Biểu đồ \(T^2\)}
  \begin{itemize}
    \item Áp dụng cho \textbf{nhiều đặc tính}, không giới hạn ở hai biến.
    \item Các điểm vẽ \textbf{theo thứ tự thời gian}, giúp thấy rõ quy luật và xu hướng.
  \end{itemize}
  \begin{block}{Thống kê cho điểm thứ \(j\)}
    \begin{equation}
      T_j^2=(\vect{x}_j-\bar{\vect{x}})\trans\mat{S}\inv(\vect{x}_j-\bar{\vect{x}})
      \tag{5-33}
    \end{equation}
  \end{block}
  \begin{itemize}
    \item LCL \(=0\); \(\mathrm{UCL}=\chi^2_p(.05)\), đôi khi \(\chi^2_p(.01)\). \textbf{Không có đường trung tâm.}
    \item \(T_j^2\) chính là \(d_j^2\) dùng kiểm định tính chuẩn ở Mục 4.6.
  \end{itemize}
\end{frame}

\begin{frame}{Khi \(T^2\) báo hiệu: biến nào gây ra?}
  Khi biểu đồ \(T^2\) báo đơn vị \(j\) ngoài tầm kiểm soát, cần xác định biến nguyên nhân. Thường dùng miền hiệu chỉnh \textbf{Bonferroni}:
  \begin{block}{Biến thứ \(k\) ngoài tầm kiểm soát nếu \(x_{jk}\) không thuộc}
    \begin{equation*}
      \Bigl(\bar{x}_k-t_{n-1}\bigl(\tfrac{.005}{p}\bigr)\sqrt{s_{kk}},\;\;
      \bar{x}_k+t_{n-1}\bigl(\tfrac{.005}{p}\bigr)\sqrt{s_{kk}}\Bigr)
    \end{equation*}
  \end{block}
  trong đó \(p\) là tổng số biến được đo.
\end{frame}

\begin{frame}{Miền kiểm soát cho quan sát tương lai}
  \begin{block}{Kết quả 5.6: miền dự báo cho quan sát tương lai \(\vect{x}\)}
    \small
    \(\vect{X}_1,\dots,\vect{X}_n\) độc lập \(N_p(\vect{\mu},\vect{\Sigma})\), \(\vect{X}\) là quan sát tương lai. Khi đó
    \begin{equation*}
      T^2=\frac{n}{n+1}(\vect{X}-\bar{\vect{X}})\trans\mat{S}\inv(\vect{X}-\bar{\vect{X}})
      \sim\frac{(n-1)p}{n-p}F_{p,n-p}
    \end{equation*}
    và ellipsoid dự báo \(100(1-\alpha)\%\) gồm mọi \(\vect{x}\) thỏa
    \begin{equation*}
      (\vect{x}-\bar{\vect{x}})\trans\mat{S}\inv(\vect{x}-\bar{\vect{x}})\le\frac{n^2-1}{n(n-p)}\,pF_{p,n-p}(\alpha)
    \end{equation*}
  \end{block}
  \footnotesize Cốt lõi: \(\operatorname{Cov}(\vect{X}-\bar{\vect{X}})=\frac{n+1}{n}\vect{\Sigma}\). Các quan sát \(\vect{x}_1,\dots,\vect{x}_n\) dùng để lập miền phải thu khi quy trình ổn định.
\end{frame}

\begin{frame}{Biểu đồ \(T^2\) cho quan sát tương lai}
  Với mỗi quan sát mới \(\vect{x}\), vẽ theo thời gian (LCL \(=0\)):
  \begin{align*}
    T^2&=\frac{n}{n+1}(\vect{x}-\bar{\vect{x}})\trans\mat{S}\inv(\vect{x}-\bar{\vect{x}})\\
    \mathrm{UCL}&=\frac{(n-1)p}{n-p}F_{p,n-p}(.05)
  \end{align*}
  Điểm trên UCL cho thấy có thể có biến động do nguyên nhân đặc biệt; cần kiểm tra quá trình để quyết định có hành động khắc phục ngay hay không.
\end{frame}

\begin{frame}{Biểu đồ dựa trên trung bình mẫu con}
  Mỗi lần lấy \(m>1\) đơn vị; có \(n\) mẫu con, trung bình \(\bar{\vect{X}}_j\), ma trận \(\mat{S}_j\), trung bình chung \(\bar{\bar{\vect{X}}}\).
  \begin{align*}
    \operatorname{Cov}(\bar{\vect{X}}_j-\bar{\bar{\vect{X}}})&=\frac{n-1}{nm}\vect{\Sigma}\\
    \mat{S}&=\frac1n(\mat{S}_1+\cdots+\mat{S}_n)
  \end{align*}
  \((nm-n)\mat{S}\) có phân phối Wishart, độc lập với \(\bar{\bar{\vect{X}}}\). Khi đó:
  \begin{equation*}
    \frac{nm}{n-1}(\bar{\vect{X}}_j-\bar{\bar{\vect{X}}})\trans\mat{S}\inv(\bar{\vect{X}}_j-\bar{\bar{\vect{X}}})
    \sim\frac{(nm-n)p}{nm-n-p+1}F_{p,\,nm-n-p+1}
    \tag{5-35}
  \end{equation*}
\end{frame}

\begin{frame}{Trung bình mẫu con: elip và \(T^2\)}
  Elip kiểm soát với \(p=2\) (vế phải thường xấp xỉ \(\chi^2_2(.05)/m\)):
  \begin{equation}
    (\bar{\vect{x}}_j-\bar{\bar{\vect{x}}})\trans\mat{S}\inv(\bar{\vect{x}}_j-\bar{\bar{\vect{x}}})\le\frac{(n-1)(m-1)\,2}{m(nm-n-1)}F_{2,nm-n-1}(.05)
    \tag{5-36}
  \end{equation}
  Biểu đồ \(T^2\), \(j=1,\dots,n\) (UCL xấp xỉ \(\chi^2_p(.05)\) khi \(n\) lớn):
  \begin{align*}
    T_j^2&=m(\bar{\vect{X}}_j-\bar{\bar{\vect{X}}})\trans\mat{S}\inv(\bar{\vect{X}}_j-\bar{\bar{\vect{X}}})\\
    \mathrm{UCL}&=\frac{(n-1)(m-1)p}{nm-n-p+1}F_{p,\,nm-n-p+1}(.05)
  \end{align*}
\end{frame}

\begin{frame}{Miền kiểm soát cho trung bình mẫu con tương lai}
  \small \(\operatorname{Cov}(\bar{\vect{X}}-\bar{\bar{\vect{X}}})=\frac{n+1}{nm}\vect{\Sigma}\), nên \(\frac{nm}{n+1}(\bar{\vect{X}}-\bar{\bar{\vect{X}}})\trans\mat{S}\inv(\bar{\vect{X}}-\bar{\bar{\vect{X}}})\) có cùng phân phối \(F\) như (5-35). Elip dự báo với \(p=2\):
  \begin{equation*}
    (\vect{x}-\bar{\bar{\vect{x}}})\trans\mat{S}\inv(\vect{x}-\bar{\bar{\vect{x}}})\le\frac{(n+1)(m-1)\,2}{m(nm-n-1)}F_{2,nm-n-1}(.05)
    \tag{5-37}
  \end{equation*}
  Biểu đồ \(T^2\):
  \begin{align*}
    T^2&=m(\bar{\vect{X}}-\bar{\bar{\vect{X}}})\trans\mat{S}\inv(\bar{\vect{X}}-\bar{\bar{\vect{X}}})\\
    \mathrm{UCL}&=\frac{(n+1)(m-1)p}{nm-n-p+1}F_{p,\,nm-n-p+1}(.05)
  \end{align*}
  Điểm ngoài elip hoặc trên UCL: quá trình đã đổi khác (tốt hoặc xấu), cần tìm nguyên nhân.
\end{frame}
