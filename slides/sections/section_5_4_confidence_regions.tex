\section{Miền tin cậy và so sánh đồng thời}
\sectionframe{Miền tin cậy và so sánh đồng thời}

\begin{frame}{Miền tin cậy và so sánh đồng thời}
  \begin{block}{Phạm vi nội dung}
    Chuyển từ kiểm định một điểm \(\boldsymbol{\mu}_0\) sang tập các giá trị \(\boldsymbol{\mu}\) còn hợp lý với dữ liệu.
  \end{block}
  \begin{itemize}
    \item Miền tin cậy đồng thời cho vectơ trung bình.
    \item So sánh đồng thời các trung bình thành phần.
    \item Cách diễn giải miền ellipsoid hoặc khoảng tin cậy Bonferroni.
  \end{itemize}
\end{frame}


\begin{frame}{Khái niệm miền tin cậy}
    Để có được phương pháp chính cho việc suy diễn từ một mẫu, ta cần mở rộng khái niệm khoảng tin cậy đơn biến thành một \textbf{miền tin cậy} đa biến. 
    
    \begin{itemize}
        \item Gọi $\theta$ là vectơ các tham số tổng thể chưa biết và $\Theta$ là tập hợp tất cả các giá trị có thể có của $\theta$. 
        \item Miền tin cậy là một miền các giá trị $\theta$ có khả năng xảy ra. Miền này được xác định bởi dữ liệu, ký hiệu là $R(X)$, trong đó $X = [X_1, X_2, \dots, X_n]'$ là ma trận dữ liệu.
    \end{itemize}

    \begin{block}{Định nghĩa}
        Miền $R(X)$ được gọi là một \textbf{miền tin cậy $100(1 - \alpha)\%$} nếu, trước khi mẫu được lấy:
        \[
            P[R(X) \text{ chứa tham số } \theta] = 1 - \alpha \quad (5-17)
        \]
        Xác suất này được tính dựa trên giá trị $\theta$ nhưng chưa biết.
    \end{block}
\end{frame}

\begin{frame}{Miền tin cậy cho phân phối chuẩn $p$-chiều}
    Miền tin cậy cho trung bình $\mu$ của một tổng thể chuẩn $p$-chiều được suy ra từ thống kê $T^2$. Trước khi mẫu được lấy:
    \[
        P\left[ n(\bar{X} - \mu)'S^{-1}(\bar{X} - \mu) \le \frac{(n - 1)p}{n - p} F_{p, n-p}(\alpha) \right] = 1 - \alpha
    \]
    bất kể giá trị chưa biết của $\mu$ và $\Sigma$ là gì. 
    
    \vspace{0.3cm}
    Nói cách khác, $\bar{X}$ sẽ nằm trong khoảng $[(n - 1)p F_{p, n-p}(\alpha)/(n - p)]^{1/2}$ so với $\mu$, với xác suất $1 - \alpha$, với điều kiện khoảng cách được đo theo $nS^{-1}$. 
\end{frame}

\begin{frame}{Phương trình Ellipsoid tin cậy}
    Với một mẫu cụ thể, $\bar{x}$ và $S$ có thể được tính toán, và bất đẳng thức
    \[
        n(\bar{x} - \mu)'S^{-1}(\bar{x} - \mu) \le \frac{(n - 1)p}{n - p} F_{p, n-p}(\alpha)
    \]
    sẽ xác định một miền $R(X)$ trong không gian của tất cả các giá trị tham số có thể có. 
    
    \vspace{0.3cm}
    Trong trường hợp này, miền này sẽ là một \textit{ellipsoid} có tâm tại $\bar{x}$.
    \begin{block}{Miền tin cậy $100(1-\alpha)\%$}
        Một miền tin cậy $100(1 - \alpha)\%$ cho trung bình của phân phối chuẩn $p$-chiều là ellipsoid được xác định bởi tất cả các $\mu$ sao cho:
        \[
            n(\bar{x} - \mu)'S^{-1}(\bar{x} - \mu) \le \frac{(n - 1)p}{n - p} F_{p, n-p}(\alpha) \quad (5-18)
        \]
    \end{block}
\end{frame}

\begin{frame}{Liên hệ với kiểm định giả thuyết}
    Trong đó $\bar{x} = \frac{1}{n} \sum_{j=1}^n x_j$ và $S = \frac{1}{n-1} \sum_{j=1}^n (x_j - \bar{x})(x_j - \bar{x})'$.
    
    \vspace{0.3cm}
    \textbf{Đánh giá giá trị $\mu_0$:}
    Để xác định xem một $\mu_0$ bất kỳ có nằm trong miền tin cậy hay không, ta cần tính \textit{khoảng cách bình phương tổng quát} $n(\bar{x} - \mu_0)'S^{-1}(\bar{x} - \mu_0)$ và so sánh nó với $\frac{p(n - 1)}{n - p} F_{p, n-p}(\alpha)$. 
    
    \begin{itemize}
        \item Nếu khoảng cách này lớn hơn, thì $\mu_0$ không nằm trong miền tin cậy.
        \item Vì điều này tương tự với việc kiểm định $H_0: \mu = \mu_0$ so với $H_1: \mu \ne \mu_0$ ở mức ý nghĩa $\alpha$, ta thấy rằng miền tin cậy (5-18) bao gồm tất cả các vectơ $\mu_0$ mà kiểm định $T^2$ \textbf{sẽ không bác bỏ} $H_0$.
    \end{itemize}
\end{frame}

\begin{frame}{Xác định các trục của Ellipsoid}
    Với $p \ge 4$, ta không thể vẽ được miền tin cậy chung cho $\mu$. Tuy nhiên, ta có thể tính các trục của ellipsoid tin cậy và độ dài tương đối của chúng. 
    
    \vspace{0.3cm}
    Các trục này được xác định từ các trị riêng $\lambda_i$ và vectơ riêng chuẩn hóa $e_i$ của $S$. Hướng và độ dài của các trục:
    \[
        n(\bar{x} - \mu)'S^{-1}(\bar{x} - \mu) \le c^2 = \frac{p(n - 1)}{n - p} F_{p, n-p}(\alpha)
    \]
    được xác định bằng cách di chuyển một khoảng
    \[
        \sqrt{\lambda_i} \cdot c / \sqrt{n} = \sqrt{\lambda_i} \cdot \sqrt{p(n - 1) F_{p, n-p}(\alpha) / [n(n - p)]}
    \]
    đơn vị dọc theo các vectơ riêng $e_i$.
\end{frame}

\begin{frame}{Xác định các trục của Ellipsoid}
    Bắt đầu từ tâm $\bar{x}$, các trục của ellipsoid tin cậy là:
    \begin{block}{}
        \[
            \pm \sqrt{\lambda_i} \cdot \sqrt{\frac{p(n - 1)}{n(n - p)} F_{p, n-p}(\alpha)} \cdot e_i
        \]
    \end{block}
    trong đó $S e_i = \lambda_i e_i, \quad i = 1, 2, \dots, p \quad (5-19)$

    \vspace{0.3cm}
    Tỷ số giữa các $\lambda_i$ sẽ giúp xác định mức độ kéo dài tương đối dọc theo các cặp trục.
\end{frame}

\begin{frame}{Ví dụ 5.3: Xây dựng ellipse tin cậy cho $\mu$}
    \textbf{Bài toán:} Dữ liệu về bức xạ lò vi sóng đã được giới thiệu. Gọi:
    \begin{itemize}
        \item $x_1 = \sqrt[4]{\text{bức xạ đo được khi đóng cửa}}$
        \item $x_2 = \sqrt[4]{\text{bức xạ đo được khi mở cửa}}$
    \end{itemize}
    
    Với $n = 42$ cặp quan sát đã biến đổi, ta tìm được rằng:
    \[
        \bar{x} = \begin{bmatrix} .564 \\ .603 \end{bmatrix}, \quad 
        S = \begin{bmatrix} .0144 & .0117 \\ .0117 & .0146 \end{bmatrix}
    \]
    \[
        S^{-1} = \begin{bmatrix} 203.018 & -163.391 \\ -163.391 & 200.228 \end{bmatrix}
    \]
\end{frame}

\begin{frame}{Ví dụ 5.3: Xây dựng ellipse tin cậy cho $\mu$}
    Giá trị riêng và vectơ riêng của $S$ là:
    \begin{align*}
        \lambda_1 &= .026, & e_1' &= [.704, .710] \\
        \lambda_2 &= .002, & e_2' &= [-.710, .704]
    \end{align*}
    
    Ellipse tin cậy $95\%$ cho $\mu$ bao gồm tất cả các giá trị $(\mu_1, \mu_2)$ thỏa mãn:
    \[
        42 [.564 - \mu_1, .603 - \mu_2] 
        \begin{bmatrix} 203.018 & -163.391 \\ -163.391 & 200.228 \end{bmatrix}
        \begin{bmatrix} .564 - \mu_1 \\ .603 - \mu_2 \end{bmatrix} \le \frac{2(41)}{40} F_{2, 40}(.05)
    \]
\end{frame}

\begin{frame}{Ví dụ 5.3: Xây dựng ellipse tin cậy cho $\mu$}
    Hay, vì $F_{2, 40}(.05) = 3.23$, nên phương trình tương đương với:
    \begin{align*}
        42(203.018)(.564 - \mu_1)^2 + 42(200.228)(.603 - \mu_2)^2 & \\
        - 84(163.391)(.564 - \mu_1)(.603 - \mu_2) &\le 6.62
    \end{align*}
    
    \textbf{Kiểm tra tính hợp lý của một điểm:}
    Để xem liệu $\mu' = [.562, .589]$ có nằm trong miền tin cậy hay không, ta tính:
    \begin{align*}
        42(203.018)(.564 - .562)^2 + 42(200.228)(.603 - .589)^2 & \\
        - 84(163.391)(.564 - .562)(.603 - .589) &= 1.30 \le 6.62
    \end{align*}
    
    Ta kết luận rằng $\mu' = [.562, .589]$ nằm trong miền tin cậy. Tương đương, một kiểm định của $H_0: \mu = [.562, .589]$ sẽ không bị bác bỏ so với $H_1: \mu \ne [.562, .589]$ ở mức $\alpha = .05$.
\end{frame}

\begin{frame}{Ví dụ 5.3: Tính toán các trục của Ellipsoid}
    Ellipsoid tin cậy chung (\textit{The joint confidence ellipsoid}) được vẽ trong Hình 5.1. Tâm của ellipsoid là $\bar{x}' = [.564, .603]$, và nửa độ dài của trục lớn và trục nhỏ được cho bởi:
    
    \textbf{Trục lớn:}
    \vspace{-0.2cm} % Thu hẹp khoảng cách với công thức
    \[
        \sqrt{\lambda_1} \cdot \sqrt{\frac{p(n - 1)}{n(n - p)} F_{p, n-p}(\alpha)} = \sqrt{.026} \cdot \sqrt{\frac{2(41)}{42(40)} (3.23)} = .064
    \]
    
    \textbf{Trục nhỏ:}
    \vspace{-0.2cm} % Thu hẹp khoảng cách với công thức
    \[
        \sqrt{\lambda_2} \cdot \sqrt{\frac{p(n - 1)}{n(n - p)} F_{p, n-p}(\alpha)} = \sqrt{.002} \cdot \sqrt{\frac{2(41)}{42(40)} (3.23)} = .018
    \]
    
    \vspace{-0.2cm} % Kéo dòng text cuối lên
    tương ứng. Các trục nằm dọc theo $e_1' = [.704, .710]$ và $e_2' = [-.710, .704]$ khi các vectơ này được vẽ với gốc tại $\bar{x}$. 
\end{frame}

\begin{frame}{Ví dụ 5.3: Mức độ kéo dài của Ellipse}
    Một chỉ số cho thấy mức độ kéo dài của ellipse tin cậy chính là tỷ số giữa độ dài trục lớn và trục nhỏ. Tỷ số này là:
    
    \[
        \frac{\sqrt{\lambda_1} \cdot \sqrt{\frac{p(n - 1)}{n(n - p)} F_{p, n-p}(\alpha)}}{\sqrt{\lambda_2} \cdot \sqrt{\frac{p(n - 1)}{n(n - p)} F_{p, n-p}(\alpha)}} = \frac{\sqrt{\lambda_1}}{\sqrt{\lambda_2}} = \frac{.161}{.045} = 3.6
    \]
\end{frame}

\begin{frame}{Nhu cầu đánh giá đồng thời các trung bình}
\begin{itemize}
  \item Miền $n(\bar{\mathbf x}-\boldsymbol\mu)'\mathbf S^{-1}(\bar{\mathbf x}-\boldsymbol\mu)\le c^2$
        cho thông tin \textbf{đồng thời, chính xác} về các giá trị hợp lý của
        $\boldsymbol\mu$.
  \item Nhưng khi báo cáo, ta thường vẫn muốn các \textbf{phát biểu riêng lẻ}
        về từng trung bình thành phần.
  \item Quan điểm: mọi phát biểu riêng lẻ đó cần \textbf{đúng đồng thời} với
        một xác suất được chỉ định cao $\Rightarrow$ đó là \textbf{khoảng tin
        cậy đồng thời} (\emph{simultaneous confidence intervals}).
\end{itemize}
\vspace{0.5em}
Ta bắt đầu từ các phát biểu tin cậy liên quan mật thiết đến miền tin cậy dựa
trên thống kê $T^2$.
\end{frame}

\begin{frame}{Phân phối của tổ hợp tuyến tính $Z=\mathbf a'\mathbf X$}
Cho $\mathbf X\sim N_p(\boldsymbol\mu,\boldsymbol\Sigma)$ và $\mathbf a$ cố định.
Đặt
\[
Z=a_1X_1+a_2X_2+\cdots+a_pX_p=\mathbf a'\mathbf X.
\]
 
Từ (2-43):
\[
\mu_Z=E(Z)=\mathbf a'\boldsymbol\mu,
\qquad
\sigma_Z^2=\operatorname{Var}(Z)=\mathbf a'\boldsymbol\Sigma\mathbf a.
\]
 
Theo Kết quả 4.2: $Z\sim N(\mathbf a'\boldsymbol\mu,\ \mathbf a'\boldsymbol\Sigma\mathbf a)$.
\end{frame}
 
% ----------------------------------------------------------------------------
\begin{frame}{Thống kê mẫu cho biến $Z$}
Với mẫu ngẫu nhiên $\mathbf X_1,\ldots,\mathbf X_n$ từ $N_p(\boldsymbol\mu,\boldsymbol\Sigma)$,
lấy tổ hợp tuyến tính cho mỗi quan sát:
\[
Z_j=a_1X_{j1}+a_2X_{j2}+\cdots+a_pX_{jp}=\mathbf a'\mathbf X_j,\qquad j=1,\ldots,n.
\]
 
Theo (3-36), trung bình và phương sai mẫu của $z_1,\ldots,z_n$ là
\[
\bar z=\mathbf a'\bar{\mathbf x},
\qquad
s_Z^2=\mathbf a'\mathbf S\mathbf a,
\]
với $\bar{\mathbf x}$, $\mathbf S$ là vector trung bình mẫu và ma trận hiệp
phương sai mẫu của $\mathbf X_j$.
 
\vspace{0.5em}
$\Rightarrow$ Có thể xây dựng khoảng tin cậy đồng thời bằng cách xét khoảng
tin cậy cho $\mathbf a'\boldsymbol\mu$ với nhiều lựa chọn khác nhau của $\mathbf a$.
\end{frame}
 
% ----------------------------------------------------------------------------
\begin{frame}{Khoảng tin cậy $t$-Student cho một tổ hợp $\mathbf{a}'\boldsymbol{\mu}$ cố định}
Với $\mathbf a$ cố định và $\sigma_Z^2$ chưa biết, dùng tỷ số $t$ Student:
\[
t=\frac{\bar z-\mu_Z}{s_Z/\sqrt n}=\frac{\sqrt n(\mathbf a'\bar{\mathbf x}-\mathbf a'\boldsymbol\mu)}{\sqrt{\mathbf a'\mathbf S\mathbf a}}
\tag{5-20}
\]
 
dẫn đến khoảng tin cậy $100(1-\alpha)\%$ cho $\mu_Z=\mathbf a'\boldsymbol\mu$:
 
\[
\mathbf a'\bar{\mathbf x}-t_{n-1}(\alpha/2)\sqrt{\frac{\mathbf a'\mathbf S\mathbf a}{n}}
\;\le\; \mathbf a'\boldsymbol\mu \;\le\;
\mathbf a'\bar{\mathbf x}+t_{n-1}(\alpha/2)\sqrt{\frac{\mathbf a'\mathbf S\mathbf a}{n}}
\tag{5-21}
\]
 
trong đó $t_{n-1}(\alpha/2)$ là bách phân vị trên $100(\alpha/2)\%$ của phân
phối $t$ với $n-1$ bậc tự do.
\end{frame}
 
% ----------------------------------------------------------------------------
\begin{frame}{Áp dụng khoảng tin cậy cho từng trung bình thành phần}
\begin{itemize}
  \item Chọn $\mathbf a'=[1,0,\ldots,0]$: $\mathbf a'\boldsymbol\mu=\mu_1$,
        $\mathbf a'\mathbf S\mathbf a=s_{11}$ $\Rightarrow$ (5-21) trở thành
        khoảng tin cậy thông thường cho \emph{một} trung bình chuẩn.
  \item Đổi $\mathbf a$, ta lập được nhiều phát biểu tin cậy về các thành
        phần của $\boldsymbol\mu$ — \textbf{mỗi phát biểu riêng lẻ} có hệ số
        tin cậy $1-\alpha$.
\end{itemize}
 
\vspace{0.5em}
\begin{block}{Nhưng}
Hệ số tin cậy gắn với \textbf{tất cả} các phát biểu đó \textbf{xét cùng nhau}
thì \textbf{không phải} là $1-\alpha$.
\end{block}
\end{frame}
 
% ----------------------------------------------------------------------------
\begin{frame}{Độ tin cậy chung và sự đánh đổi độ rộng khoảng tin cậy}
\begin{itemize}
  \item Về trực giác, muốn có một \textbf{``hệ số tin cậy chung''} $1-\alpha$
        cho các khoảng tin cậy sinh ra bởi \textbf{mọi} lựa chọn của $\mathbf a$
        — đây chính là sự đảm bảo về xác suất cho việc \emph{bất kỳ} phát
        biểu nào sai.
  \item \textbf{Cái giá phải trả:} các khoảng tin cậy đồng thời sẽ
        \textbf{rộng hơn (kém chính xác hơn)} so với khoảng (5-21) ứng với
        \emph{một} lựa chọn cụ thể của $\mathbf a$.
\end{itemize}
 
\vspace{0.5em}
$\Rightarrow$ Cần một hằng số lớn hơn $t^2_{n-1}(\alpha/2)$ áp dụng cho mọi
$\mathbf a$ cùng lúc — dẫn tới Kết quả 5.3 (khung tiếp theo).
\end{frame}
\begin{frame}{Từ So Sánh Đơn Lẻ Đến Đánh Giá Đồng Thời Cho Mọi Vectơ $a$}
Với dữ liệu $\mathbf x_1,\ldots,\mathbf x_n$ và một $\mathbf a$ cụ thể, khoảng tin cậy
$t$ trong (5-21) chính là tập các giá trị $\mathbf a'\boldsymbol\mu$ thoả

\[
t^2=\frac{n\bigl(\mathbf a'\bar{\mathbf x}-\mathbf a'\boldsymbol\mu\bigr)^2}{\mathbf a'\mathbf S\mathbf a}
\;\le\; t^2_{n-1}(\alpha/2)
\tag{5-22}
\]

\begin{itemize}
  \item Muốn phát biểu \textbf{đồng thời} cho \emph{nhiều} lựa chọn $\mathbf a$,
        hằng số $t^2_{n-1}(\alpha/2)$ phải được thay bằng một hằng số lớn hơn, $c^2$.
  \item Xét các $\mathbf a$ sao cho $t^2\le c^2$: điều này tự nhiên dẫn tới việc
        tìm
\[
\max_{\mathbf a}\, t^2=\max_{\mathbf a}\frac{n\bigl[\mathbf a'(\bar{\mathbf x}-\boldsymbol\mu)\bigr]^2}{\mathbf a'\mathbf S\mathbf a}.
\]
\end{itemize}
\end{frame}

% ----------------------------------------------------------------------------
\begin{frame}{Ứng Dụng Bổ Đề Cực Đại Hóa Thiết Lập Thống Kê $T^2$}
Áp dụng bổ đề cực đại hoá (2-50) với $\mathbf x=\mathbf a$, $\mathbf d=(\bar{\mathbf x}-\boldsymbol\mu)$, $\mathbf B=\mathbf S$:

\begin{keyformula}[Liên hệ với thống kê Hotelling $T^2$]
{\small
\[
\max_{\mathbf a}\frac{n\bigl[\mathbf a'(\bar{\mathbf x}-\boldsymbol\mu)\bigr]^2}{\mathbf a'\mathbf S\mathbf a}
= n\Bigl[\max_{\mathbf a}\frac{\bigl[\mathbf a'(\bar{\mathbf x}-\boldsymbol\mu)\bigr]^2}{\mathbf a'\mathbf S\mathbf a}\Bigr]
= n(\bar{\mathbf x}-\boldsymbol\mu)'\mathbf S^{-1}(\bar{\mathbf x}-\boldsymbol\mu)
= T^2
\tag{5-23}
\]
}
\end{keyformula}

\begin{itemize}
  \item Giá trị cực đại đạt được khi $\mathbf a$ \textbf{tỉ lệ với} $\mathbf S^{-1}(\bar{\mathbf x}-\boldsymbol\mu)$.
  \item Đây chính là thống kê Hotelling $T^2$ đã biết — không phải một đại lượng mới.
\end{itemize}
\end{frame}

% ----------------------------------------------------------------------------
\begin{frame}{Thiết Lập Khoảng Tin Cậy Đồng Thời}
\textbf{Kết quả 5.3.} Cho $\mathbf X_1,\mathbf X_2,\ldots,\mathbf X_n$ là một mẫu
ngẫu nhiên từ tổng thể $N_p(\boldsymbol\mu,\boldsymbol\Sigma)$ với $\boldsymbol\Sigma$
xác định dương. Khi đó, \textbf{đồng thời với mọi $\mathbf a$}, khoảng

\begin{keyformula}[Khoảng tin cậy đồng thời]
{\small
\[
\left( \mathbf{a}'\bar{\mathbf{X}} - \sqrt{\frac{p(n-1)}{n-p}F_{p,n-p}(\alpha)} \sqrt{\frac{\mathbf{a}'\mathbf{S}\mathbf{a}}{n}} \; , \; \mathbf{a}'\bar{\mathbf{X}} + \sqrt{\frac{p(n-1)}{n-p}F_{p,n-p}(\alpha)} \sqrt{\frac{\mathbf{a}'\mathbf{S}\mathbf{a}}{n}} \right)
\]}
\end{keyformula}

sẽ chứa $\mathbf a'\boldsymbol\mu$ với xác suất $1-\alpha$.

\vspace{0.5em}
\begin{itemize}
  \item Khác với (5-22): hằng số $t^2_{n-1}(\alpha/2)$ được thay bằng
        $c^2=\dfrac{p(n-1)}{n-p}F_{p,n-p}(\alpha)$ để giữ đúng xác suất
        \textbf{cho mọi $\mathbf a$ cùng lúc}.
\end{itemize}
\end{frame}

% ----------------------------------------------------------------------------
\begin{frame}{Chứng minh Kết quả 5.3}
Từ (5-23): $T^2=n(\bar{\mathbf x}-\boldsymbol\mu)'\mathbf S^{-1}(\bar{\mathbf x}-\boldsymbol\mu)\le c^2$
kéo theo, \textbf{với mọi $\mathbf a$},

\[
\frac{n(\mathbf a'\bar{\mathbf x}-\mathbf a'\boldsymbol\mu)^2}{\mathbf a'\mathbf S\mathbf a}\le c^2
\quad\Longleftrightarrow\quad
\mathbf a'\bar{\mathbf x}-c\sqrt{\frac{\mathbf a'\mathbf S\mathbf a}{n}}
\le \mathbf a'\boldsymbol\mu \le
\mathbf a'\bar{\mathbf x}+c\sqrt{\frac{\mathbf a'\mathbf S\mathbf a}{n}}.
\]

\[
c^2=\frac{p(n-1)}{n-p}F_{p,n-p}(\alpha),
\qquad
P\!\left(T^2\le c^2\right)=1-\alpha.
\]

Do đó, các khoảng trên đồng thời chứa $\mathbf a'\boldsymbol\mu$ với mọi
$\mathbf a$, với xác suất $1-\alpha$.

\end{frame}


% ----------------------------------------------------------------------------
\begin{frame}{Các khoảng $T^2$ cho từng thành phần $\mu_i$}
Gọi các khoảng đồng thời của Kết quả 5.3 là \textbf{các khoảng $T^2$}, vì xác
suất phủ được xác định bởi phân phối của $T^2$.

Lần lượt chọn $\mathbf a'=[1,0,\ldots,0],\ [0,1,\ldots,0],\ \ldots,\ [0,\ldots,0,1]$:

\begin{keyformula}[Khoảng cho từng thành phần]
{\footnotesize
\begin{equation*}
\tag{5-24}
\begin{aligned}
\bar x_1-\sqrt{\tfrac{p(n-1)}{n-p}F_{p,n-p}(\alpha)}\sqrt{\tfrac{s_{11}}{n}}
&\le \mu_1\le
\bar x_1+\sqrt{\tfrac{p(n-1)}{n-p}F_{p,n-p}(\alpha)}\sqrt{\tfrac{s_{11}}{n}}\\
&\ \vdots\\
\bar x_p-\sqrt{\tfrac{p(n-1)}{n-p}F_{p,n-p}(\alpha)}\sqrt{\tfrac{s_{pp}}{n}}
&\le \mu_p\le
\bar x_p+\sqrt{\tfrac{p(n-1)}{n-p}F_{p,n-p}(\alpha)}\sqrt{\tfrac{s_{pp}}{n}}
\end{aligned}
\end{equation*}}
\end{keyformula}

đều đồng thời có độ tin cậy đạt ít nhất $1-\alpha$.
\end{frame}

\begin{frame}{Mở Rộng: Khoảng Tin Cậy Đồng Thời Cho Hiệu Hai Trung Bình}
  \textbf{Khoảng tin cậy cho hiệu $\mu_i - \mu_k$:}
  
  Chọn $\mathbf{a} = [0, \dots, 1, \dots, -1, \dots, 0]'$ sao cho $a_i = 1$, $a_k = -1$ và các thành phần khác bằng $0$. Khi đó:
  \[
    \mathbf{a}'\mathbf{S}\mathbf{a} = s_{ii} - 2s_{ik} + s_{kk}
  \]
  
  \vspace{-0.5em}
  Hệ số tin cậy $1-\alpha$ được bảo đảm không đổi theo công thức:
  \begin{equation*}
    (\bar{x}_i - \bar{x}_k) - c \sqrt{\frac{s_{ii} - 2s_{ik} + s_{kk}}{n}}
    \le \mu_i - \mu_k \le
    (\bar{x}_i - \bar{x}_k) + c \sqrt{\frac{s_{ii} - 2s_{ik} + s_{kk}}{n}}
    \tag{5-25}
  \end{equation*}
  với hằng số tới hạn $c = \sqrt{\frac{p(n-1)}{n-p} F_{p,n-p}(\alpha)}$.

  \begin{itemize}
    \item Đây là trường hợp riêng của khoảng đồng thời $T^2$ cho tổ hợp $\mathbf{a}'\boldsymbol{\mu}$.
    \item Độ tin cậy chung $1-\alpha$ luôn được bảo đảm mà không cần hiệu chỉnh Bonferroni.
  \end{itemize}
\end{frame}

% ----------------------------------------------------------------------------
\begin{frame}{Mở Rộng: Elip Tin Cậy Cho Cặp Trung Bình}
  \begin{block}{Elip tin cậy cho cặp $(\mu_i, \mu_k)$ (Phụ lục 5A)}
    Tập hợp các giá trị $(\mu_i, \mu_k)$ thỏa mãn bất đẳng thức:
    \begin{equation*}
      n
      \begin{bmatrix}
        \bar{x}_i - \mu_i \\
        \bar{x}_k - \mu_k
      \end{bmatrix}'
      \begin{bmatrix}
        s_{ii} & s_{ik} \\
        s_{ik} & s_{kk}
      \end{bmatrix}^{-1}
      \begin{bmatrix}
        \bar{x}_i - \mu_i \\
        \bar{x}_k - \mu_k
      \end{bmatrix}
      \le \frac{p(n-1)}{n-p} F_{p,n-p}(\alpha) \tag{5-26}
    \end{equation*}
  \end{block}

  \begin{itemize}
    \item \textbf{Khảo sát linh hoạt:} Do độ tin cậy $1-\alpha$ giữ nguyên với mọi $\mathbf{a}$, ta có thể lập khoảng $T^2$ cho bất kỳ tổ hợp tuyến tính nào sau khi quan sát dữ liệu.
    \item \textbf{Ý nghĩa hình học:} Miền elip mô tả đồng thời miền khả dĩ của $(\mu_i, \mu_k)$, tích hợp cả biến động biên ($s_{ii}, s_{kk}$) lẫn tương quan mẫu ($s_{ik}$).
  \end{itemize}
\end{frame}

\begin{frame}{Phương pháp so sánh riêng lẻ từng biến}
Bỏ qua cấu trúc hiệp phương sai, chỉ dùng $\mathbf a'=[0,\ldots,a_i,\ldots,0]$
với $a_i=1$ (gợi ý bởi (5-21)):

{\footnotesize
\begin{equation*}
\tag{5-27}
\begin{aligned}
\bar x_1-t_{n-1}(\alpha/2)\sqrt{\tfrac{s_{11}}{n}}&\le \mu_1\le
\bar x_1+t_{n-1}(\alpha/2)\sqrt{\tfrac{s_{11}}{n}}\\
&\ \vdots\\
\bar x_p-t_{n-1}(\alpha/2)\sqrt{\tfrac{s_{pp}}{n}}&\le \mu_p\le
\bar x_p+t_{n-1}(\alpha/2)\sqrt{\tfrac{s_{pp}}{n}}
\end{aligned}
\end{equation*}}

\begin{itemize}
  \item Trước khi lấy mẫu, khoảng thứ $i$ có xác suất $1-\alpha$ phủ $\mu_i$.
  \item Nhưng xác suất để \textbf{tất cả} các khoảng cùng chứa các $\mu_i$
        tương ứng \textbf{không phải là} $1-\alpha$ nói chung.
\end{itemize}

\begin{frame}{Hiện tượng sụt giảm độ tin cậy tổng thể khi các biến độc lập}
Xét trường hợp đặc biệt: phân phối chuẩn đồng thời và
\[
\boldsymbol\Sigma=\operatorname{diag}(\sigma_{11},\sigma_{22},\ldots,\sigma_{pp}).
\]
Khi đó các biến độc lập $\Rightarrow$ áp dụng quy tắc nhân xác suất:
\[
P[\text{tất cả khoảng $t$ trong (5-27) đều chứa }\mu_i]
=(1-\alpha)(1-\alpha)\cdots(1-\alpha)=(1-\alpha)^p
\]

\begin{block}{Ví dụ số}
$1-\alpha=0{,}95,\ p=6$ $\Rightarrow$ $(0{,}95)^6\approx 0{,}74$.
\end{block}

Dù mỗi khoảng ``tin cậy 95\%'', khẳng định \textbf{chung} cho cả 6 biến chỉ còn
đúng khoảng 74\%.
\end{frame}

\begin{frame}{Sự gia tăng độ rộng của khoảng tin cậy đồng thời $T^2$}
Để giữ xác suất $1-\alpha$ cho \textbf{tất cả} phát biểu cùng lúc, khoảng cho
từng thành phần phải rộng hơn khoảng $t$ riêng lẻ — mức độ phụ thuộc $p$, $n$,
$1-\alpha$.

\begin{block}{Ví dụ với: $1-\alpha=0{,}95,\ n=15,\ p=4$}
\[

\qquad\text{so với}\qquad
t_{14}(.025)=2{,}145
\]
\[
100\left(\frac{4{,}14-2{,}145}{2{,}145}\right)\approx 93\%\ \text{rộng hơn}
\]
\end{block}
\end{frame}

% ----------------------------------------------------------------------------
\begin{frame}{Đánh giá phương pháp $T^2$ \& động lực cho phương pháp Bonferroni}
\begin{itemize}
  \item Mức tin cậy của \textbf{tập} khoảng $T^2$ luôn là $0{,}95$ (với $n,p$
        cố định); còn mức tin cậy \textbf{tổng thể} của tập khoảng $t$ riêng
        lẻ có thể nhỏ hơn $0{,}95$ rất nhiều $\Rightarrow$ hai bên không cùng
        ``mặt bằng''.
  \item Khoảng $t$ riêng lẻ vẫn là thông tin tốt nhất cho \emph{một} trung
        bình, nếu đó là suy luận duy nhất; một số nhà nghiên cứu chỉ tính
        chúng khi kiểm định $T^2$ đã bác bỏ $H_0$.
\end{itemize}

\vspace{0.5em}
\textbf{Ngược lại,} nếu \emph{chỉ} quan tâm $p$ trung bình thành phần, khoảng
$T^2$ lại \textbf{quá rộng}:
\begin{itemize}
  \item Nếu $\mu_1,\mu_2$ nằm trong khoảng $T^2$ của chúng thì $(\mu_1,\mu_2)$
        nằm trong hình chữ nhật tạo bởi hai khoảng đó — hình chữ nhật này
        \textbf{chứa elip tin cậy và còn hơn thế}.
  \item Elip (nhỏ hơn hình chữ nhật) đã có xác suất đúng $0{,}95$ phủ
        $\boldsymbol\mu$ $\Rightarrow$ xác suất phủ $(\mu_1,\mu_2)$ của hình
        chữ nhật \textbf{lớn hơn} $0{,}95$: độ tin cậy ``thừa'' không cần thiết.
\end{itemize}

$\Rightarrow$ Động lực cho cách tiếp cận thứ hai: \textbf{phương pháp Bonferroni}.
\end{frame}

% Mục 5.4 - Phần 3: Phương pháp so sánh bội Bonferroni và phần kết
\begin{frame}{Đặt vấn đề}
  \begin{itemize}
    \item Khoảng $T^2$ quá rộng nếu chỉ áp dụng cho riêng $p$ trung bình thành phần.
    \item Xét Hình 5.2: nếu $\mu_1$ nằm trong khoảng $T^2$ của nó và $\mu_2$ nằm trong khoảng $T^2$ của nó thì $(\mu_1,\mu_2)$ nằm trong hình chữ nhật tạo bởi hai khoảng này.
    \item Hình chữ nhật này \textbf{chứa elip tin cậy và còn hơn thế nữa}: xác suất phủ đồng thời $\mu_1,\mu_2$ lớn hơn $0{,}95$.
    \item Thông thường, ta chỉ quan tâm một \textbf{số nhỏ} $m$ phát biểu tin cậy $\Rightarrow$ có thể xây dựng khoảng đồng thời \textbf{ngắn hơn (chính xác hơn)} khoảng $T^2$.
    \item Cách làm này gọi là \textbf{phương pháp Bonferroni}, vì xuất phát từ một bất đẳng thức xác suất cùng tên.
  \end{itemize}
  \note{Câu hỏi có thể gặp: \emph{Vì sao $T^2$ quá rộng?} Vì khoảng $T^2$ đảm bảo cho mọi $\mathbf{a}$, còn ta chỉ cần $p$ trung bình. Hình chữ nhật tạo bởi các khoảng $T^2$ chứa elip tin cậy (elip có xác suất phủ đúng $0{,}95$) nên xác suất phủ của hình chữ nhật lớn hơn $0{,}95$, tức là ``dư'' độ tin cậy so với mức cần.}
\end{frame}
 
% ---------------------------------------------------------------------
\begin{frame}{Thiết lập bài toán}
  \begin{itemize}
    \item \textbf{Trước khi thu thập dữ liệu}, giả sử cần các phát biểu tin cậy về $m$ tổ hợp tuyến tính:
    \[
      \mathbf{a}_1'\boldsymbol{\mu},\; \mathbf{a}_2'\boldsymbol{\mu},\; \ldots,\; \mathbf{a}_m'\boldsymbol{\mu},
      \qquad \mathbf{a}'\boldsymbol{\mu} = a_1\mu_1 + a_2\mu_2 + \cdots + a_p\mu_p
    \]
    \item Gọi $C_i$ là phát biểu tin cậy về giá trị của $\mathbf{a}_i'\boldsymbol{\mu}$, với
    \[
      P[C_i \text{ đúng}] = 1-\alpha_i, \qquad i = 1,2,\ldots,m
    \]
    \item Mục tiêu: kiểm soát xác suất để \textbf{tất cả} $C_i$ cùng đúng.
  \end{itemize}
  \note{Câu hỏi có thể gặp: \emph{Vì sao phải chọn phát biểu trước khi thu thập dữ liệu?} Sách đặt giả thiết ``prior to the collection of data'' khi xây dựng phương pháp. Ngược lại, các khoảng $T^2$ ``lý tưởng cho dò quét dữ liệu'' vì hệ số tin cậy $1-\alpha$ không đổi với mọi cách chọn $\mathbf{a}$ (sách trang 226).}
\end{frame}
 
% ---------------------------------------------------------------------
\begin{frame}{Bất đẳng thức Bonferroni }
  \small
  \setlength{\abovedisplayskip}{2pt}
  \setlength{\belowdisplayskip}{2pt}
  \begin{align*}
    P[\text{tất cả } C_i \text{ đúng}] &= 1 - P[\text{ít nhất một } C_i \text{ sai}] \\
    &\ge 1 - \sum_{i=1}^m P(C_i \text{ sai}) = 1 - \sum_{i=1}^m \bigl(1 - P(C_i \text{ đúng})\bigr) \\
    &= 1 - (\alpha_1 + \alpha_2 + \cdots + \alpha_m) \tag{5-28}
  \end{align*}
  \footnotesize
  \begin{itemize}
    \item Cho phép kiểm soát \textbf{tỷ lệ sai số tổng thể} $\alpha_1+\cdots+\alpha_m$, \textbf{bất kể cấu trúc tương quan} đằng sau các phát biểu.
    \item Có thể kiểm soát tỷ lệ sai số cho một nhóm phát biểu quan trọng, và cân bằng bằng một lựa chọn khác cho các phát biểu ít quan trọng hơn.
  \end{itemize}
  \note{Câu hỏi có thể gặp: \emph{Bước dấu $\ge$ từ đâu?} Biến cố ``ít nhất một $C_i$ sai'' là hợp của các biến cố ``$C_i$ sai'', và xác suất của một hợp không vượt quá tổng các xác suất, nên $P[\text{ít nhất một sai}] \le \sum P(C_i \text{ sai})$; trừ hai vế từ $1$ ta được dấu $\ge$. Vì chỉ dùng tổng xác suất nên không cần biết các $C_i$ độc lập hay tương quan. Bước tiếp theo dùng $P(C_i \text{ sai}) = 1 - P(C_i \text{ đúng}) = \alpha_i$.}
\end{frame}
 
% ---------------------------------------------------------------------
\begin{frame}{Khoảng Bonferroni cho các trung bình thành phần}
  \begin{itemize}
    \item Không có thông tin về tầm quan trọng tương đối của các thành phần $\mu_i$ $\Rightarrow$ xét các khoảng $t$ riêng lẻ với $\alpha_i = \alpha/m$:
    \[
      \bar{x}_i \pm t_{n-1}\!\left(\frac{\alpha_i}{2}\right)\sqrt{\frac{s_{ii}}{n}}, \quad i=1,\ldots,m
    \]
    \item Vì $P\!\left[\bar{X}_i \pm t_{n-1}\!\left(\frac{\alpha}{2m}\right)\sqrt{\frac{S_{ii}}{n}} \text{ chứa } \mu_i\right] = 1-\frac{\alpha}{m}$, từ (5-28):
    \[
      P\!\left[\bar{X}_i \pm t_{n-1}\!\left(\tfrac{\alpha}{2m}\right)\sqrt{\tfrac{S_{ii}}{n}} \text{ chứa } \mu_i,\ \forall i\right]
      \ge 1 - \underbrace{\left(\tfrac{\alpha}{m}+\tfrac{\alpha}{m}+\cdots+\tfrac{\alpha}{m}\right)}_{m \text{ số hạng}} = 1-\alpha
    \]
  \end{itemize}
  \note{Câu hỏi có thể gặp: \emph{Vì sao là $t_{n-1}(\alpha/2m)$?} Khoảng $t$ hai phía có mức ý nghĩa $\alpha_i=\alpha/m$ dùng điểm phần trăm trên $100(\alpha_i/2)$ của phân phối $t$ với $n-1$ bậc tự do, tức $t_{n-1}\!\left(\frac{\alpha}{2m}\right)$ (cách định nghĩa $t_{n-1}(\alpha/2)$ ở trang 224). \emph{Vì sao chọn $\alpha_i=\alpha/m$?} Vì không biết thành phần nào quan trọng hơn, và $m$ số hạng $\alpha/m$ cộng lại đúng bằng $\alpha$.}
\end{frame}
 
% ---------------------------------------------------------------------
\begin{frame}{Khoảng Bonferroni và so sánh với khoảng $T^2$}
  Với mức tin cậy tổng thể $\ge 1-\alpha$, ta đưa ra $m=p$ phát biểu:
  \begin{equation*}
    \bar{x}_i - t_{n-1}\!\left(\frac{\alpha}{2p}\right)\sqrt{\frac{s_{ii}}{n}} \;\le\; \mu_i \;\le\; \bar{x}_i + t_{n-1}\!\left(\frac{\alpha}{2p}\right)\sqrt{\frac{s_{ii}}{n}}, \quad i=1,\dots,p \tag{5-29}
  \end{equation*}
  \begin{block}{So sánh (5-29) với (5-24)}
    Giá trị $t_{n-1}\!\left(\dfrac{\alpha}{2p}\right)$ thay cho $\sqrt{\dfrac{(n-1)\,p}{n-p}F_{p,n-p}(\alpha)}$; ngoài điểm thay thế này, các khoảng có \textbf{cùng cấu trúc}.
  \end{block}
  \note{Câu hỏi có thể gặp: \emph{Khác biệt duy nhất giữa hai phương pháp?} Chỉ ở hệ số nhân của $\sqrt{s_{ii}/n}$: $T^2$ dùng $\sqrt{p(n-1)F_{p,n-p}(\alpha)/(n-p)}$, Bonferroni dùng $t_{n-1}(\alpha/2p)$. Hai hệ số này được so sánh bằng số ở Bảng 5.4 (slide sau).}
\end{frame}

\begin{frame}{So sánh độ dài: Bonferroni và $T^2$}
  \small
  \setlength{\abovedisplayskip}{2pt}
  \setlength{\belowdisplayskip}{2pt}

          \item Kiểm soát tỷ lệ sai số tổng thể $\alpha_1+\cdots+\alpha_m$, \textbf{bất kể cấu trúc tương quan}.
          \item Linh hoạt chia $\alpha_i$: nhóm phát biểu quan trọng dùng $\alpha_i$ khác với nhóm ít quan trọng.
          \item Với $m$ nhỏ, khoảng \textbf{ngắn hơn (chính xác hơn)} khoảng $T^2$ (Bảng 5.4).
          \item Dễ áp dụng.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Phạm vi áp dụng}
        \small
        \begin{itemize}
          \item Chú ý đến một \textbf{số nhỏ} $m$ phát biểu tin cậy.
          \item Các phát biểu được \textbf{chỉ định trước khi thu thập dữ liệu}.
          \item Nếu muốn chọn $\mathbf{a}$ sau khi xem dữ liệu (dò quét dữ liệu), khoảng $T^2$ là lý tưởng vì $1-\alpha$ không đổi với mọi $\mathbf{a}$.
          \item Bonferroni cho khoảng từng thành phần với độ tin cậy chung ≥ 1−α bất kể cấu trúc tương quan; để xét cặp $(\mu_i,\mu_k)$ có tính đến tương quan thì dùng elip tin cậy.
        \end{itemize}
      \end{block}
    \end{column}

\begin{frame}{Từ So Sánh Đơn Lẻ Đến Đánh Giá Đồng Thời Cho Mọi Vectơ $a$}
Với dữ liệu $\mathbf x_1,\ldots,\mathbf x_n$ và một $\mathbf a$ cụ thể, khoảng tin cậy
$t$ trong (5-21) chính là tập các giá trị $\mathbf a'\boldsymbol\mu$ thoả

\[
t^2=\frac{n\bigl(\mathbf a'\bar{\mathbf x}-\mathbf a'\boldsymbol\mu\bigr)^2}{\mathbf a'\mathbf S\mathbf a}
\;\le\; t^2_{n-1}(\alpha/2)
\tag{5-22}
\]

\begin{itemize}
  \item Muốn phát biểu \textbf{đồng thời} cho \emph{nhiều} lựa chọn $\mathbf a$,
        hằng số $t^2_{n-1}(\alpha/2)$ phải được thay bằng một hằng số lớn hơn, $c^2$.
  \item Xét các $\mathbf a$ sao cho $t^2\le c^2$: điều này tự nhiên dẫn tới việc
        tìm
\[
\max_{\mathbf a}\, t^2=\max_{\mathbf a}\frac{n\bigl[\mathbf a'(\bar{\mathbf x}-\boldsymbol\mu)\bigr]^2}{\mathbf a'\mathbf S\mathbf a}.
\]
\end{itemize}
\end{frame}

\begin{frame}{Ứng Dụng Bổ Đề Cực Đại Hóa Thiết Lập Thống Kê $T^2$}
Áp dụng bổ đề cực đại hoá (2-50) với $\mathbf x=\mathbf a$, $\mathbf d=(\bar{\mathbf x}-\boldsymbol\mu)$, $\mathbf B=\mathbf S$:

\begin{keyformula}[Liên hệ với thống kê Hotelling $T^2$]
{\small
\[
\max_{\mathbf a}\frac{n\bigl[\mathbf a'(\bar{\mathbf x}-\boldsymbol\mu)\bigr]^2}{\mathbf a'\mathbf S\mathbf a}
= n\Bigl[\max_{\mathbf a}\frac{\bigl[\mathbf a'(\bar{\mathbf x}-\boldsymbol\mu)\bigr]^2}{\mathbf a'\mathbf S\mathbf a}\Bigr]
= n(\bar{\mathbf x}-\boldsymbol\mu)'\mathbf S^{-1}(\bar{\mathbf x}-\boldsymbol\mu)
= T^2
\tag{5-23}
\]
}
\end{keyformula}

\begin{itemize}
  \item Giá trị cực đại đạt được khi $\mathbf a$ \textbf{tỉ lệ với} $\mathbf S^{-1}(\bar{\mathbf x}-\boldsymbol\mu)$.
  \item Đây chính là thống kê Hotelling $T^2$ đã biết — không phải một đại lượng mới.
\end{itemize}
\end{frame}

% ----------------------------------------------------------------------------
\begin{frame}{Thiết Lập Khoảng Tin Cậy Đồng Thời}
\textbf{Kết quả 5.3.} Cho $\mathbf X_1,\mathbf X_2,\ldots,\mathbf X_n$ là một mẫu
ngẫu nhiên từ tổng thể $N_p(\boldsymbol\mu,\boldsymbol\Sigma)$ với $\boldsymbol\Sigma$
xác định dương. Khi đó, \textbf{đồng thời với mọi $\mathbf a$}, khoảng

\begin{keyformula}[Khoảng tin cậy đồng thời]
{\small
\[
\left( \mathbf{a}'\bar{\mathbf{X}} - \sqrt{\frac{p(n-1)}{n-p}F_{p,n-p}(\alpha)} \sqrt{\frac{\mathbf{a}'\mathbf{S}\mathbf{a}}{n}} \; , \; \mathbf{a}'\bar{\mathbf{X}} + \sqrt{\frac{p(n-1)}{n-p}F_{p,n-p}(\alpha)} \sqrt{\frac{\mathbf{a}'\mathbf{S}\mathbf{a}}{n}} \right)
\]}
\end{keyformula}

sẽ chứa $\mathbf a'\boldsymbol\mu$ với xác suất $1-\alpha$.

\vspace{0.5em}
\begin{itemize}
  \item Khác với (5-22): hằng số $t^2_{n-1}(\alpha/2)$ được thay bằng
        $c^2=\dfrac{p(n-1)}{n-p}F_{p,n-p}(\alpha)$ để giữ đúng xác suất
        \textbf{cho mọi $\mathbf a$ cùng lúc}.
\end{itemize}
\end{frame}

% ----------------------------------------------------------------------------
\begin{frame}{Chứng minh Kết quả 5.3}
Từ (5-23): $T^2=n(\bar{\mathbf x}-\boldsymbol\mu)'\mathbf S^{-1}(\bar{\mathbf x}-\boldsymbol\mu)\le c^2$
kéo theo, \textbf{với mọi $\mathbf a$},

\[
\frac{n(\mathbf a'\bar{\mathbf x}-\mathbf a'\boldsymbol\mu)^2}{\mathbf a'\mathbf S\mathbf a}\le c^2
\quad\Longleftrightarrow\quad
\mathbf a'\bar{\mathbf x}-c\sqrt{\frac{\mathbf a'\mathbf S\mathbf a}{n}}
\le \mathbf a'\boldsymbol\mu \le
\mathbf a'\bar{\mathbf x}+c\sqrt{\frac{\mathbf a'\mathbf S\mathbf a}{n}}.
\]

\[
c^2=\frac{p(n-1)}{n-p}F_{p,n-p}(\alpha),
\qquad
P\!\left(T^2\le c^2\right)=1-\alpha.
\]

Do đó, các khoảng trên đồng thời chứa $\mathbf a'\boldsymbol\mu$ với mọi
$\mathbf a$, với xác suất $1-\alpha$.

\end{frame}

\subsection{Các khoảng \texorpdfstring{$T^2$}{T²} đồng thời và ý nghĩa hình học}

% ----------------------------------------------------------------------------
\begin{frame}{Các khoảng $T^2$ cho từng thành phần $\mu_i$}
Gọi các khoảng đồng thời của Kết quả 5.3 là \textbf{các khoảng $T^2$}, vì xác
suất phủ được xác định bởi phân phối của $T^2$.

Lần lượt chọn $\mathbf a'=[1,0,\ldots,0],\ [0,1,\ldots,0],\ \ldots,\ [0,\ldots,0,1]$:

\begin{keyformula}[Khoảng cho từng thành phần]
{\footnotesize
\begin{equation*}
\tag{5-24}
\begin{aligned}
\bar x_1-\sqrt{\tfrac{p(n-1)}{n-p}F_{p,n-p}(\alpha)}\sqrt{\tfrac{s_{11}}{n}}
&\le \mu_1\le
\bar x_1+\sqrt{\tfrac{p(n-1)}{n-p}F_{p,n-p}(\alpha)}\sqrt{\tfrac{s_{11}}{n}}\\
&\ \vdots\\
\bar x_p-\sqrt{\tfrac{p(n-1)}{n-p}F_{p,n-p}(\alpha)}\sqrt{\tfrac{s_{pp}}{n}}
&\le \mu_p\le
\bar x_p+\sqrt{\tfrac{p(n-1)}{n-p}F_{p,n-p}(\alpha)}\sqrt{\tfrac{s_{pp}}{n}}
\end{aligned}
\end{equation*}}
\end{keyformula}

đều đồng thời có độ tin cậy đạt ít nhất $1-\alpha$.
\end{frame}

% ----------------------------------------------------------------------------
% ----------------------------------------------------------------------------
\begin{frame}{Mở Rộng: Khoảng Tin Cậy Đồng Thời Cho Hiệu Hai Trung Bình}
  \textbf{Khoảng tin cậy cho hiệu $\mu_i - \mu_k$:}
  
  Chọn $\mathbf{a} = [0, \dots, 1, \dots, -1, \dots, 0]'$ sao cho $a_i = 1$, $a_k = -1$ và các thành phần khác bằng $0$. Khi đó:
  \[
    \mathbf{a}'\mathbf{S}\mathbf{a} = s_{ii} - 2s_{ik} + s_{kk}
  \]
  
  \vspace{-0.5em}
  Hệ số tin cậy $1-\alpha$ được bảo đảm không đổi theo công thức:
  \begin{equation*}
    (\bar{x}_i - \bar{x}_k) - c \sqrt{\frac{s_{ii} - 2s_{ik} + s_{kk}}{n}}
    \le \mu_i - \mu_k \le
    (\bar{x}_i - \bar{x}_k) + c \sqrt{\frac{s_{ii} - 2s_{ik} + s_{kk}}{n}}
    \tag{5-25}
  \end{equation*}
  với hằng số tới hạn $c = \sqrt{\frac{p(n-1)}{n-p} F_{p,n-p}(\alpha)}$.

  \begin{itemize}
    \item Đây là trường hợp riêng của khoảng đồng thời $T^2$ cho tổ hợp $\mathbf{a}'\boldsymbol{\mu}$.
    \item Độ tin cậy chung $1-\alpha$ luôn được bảo đảm mà không cần hiệu chỉnh Bonferroni.
  \end{itemize}
\end{frame}

% ----------------------------------------------------------------------------
\begin{frame}{Mở Rộng: Elip Tin Cậy Cho Cặp Trung Bình}
  \begin{block}{Elip tin cậy cho cặp $(\mu_i, \mu_k)$ (Phụ lục 5A)}
    Tập hợp các giá trị $(\mu_i, \mu_k)$ thỏa mãn bất đẳng thức:
    \begin{equation*}
      n
      \begin{bmatrix}
        \bar{x}_i - \mu_i \\
        \bar{x}_k - \mu_k
      \end{bmatrix}'
      \begin{bmatrix}
        s_{ii} & s_{ik} \\
        s_{ik} & s_{kk}
      \end{bmatrix}^{-1}
      \begin{bmatrix}
        \bar{x}_i - \mu_i \\
        \bar{x}_k - \mu_k
      \end{bmatrix}
      \le \frac{p(n-1)}{n-p} F_{p,n-p}(\alpha) \tag{5-26}
    \end{equation*}
  \end{block}

  \begin{itemize}
    \item \textbf{Khảo sát linh hoạt:} Do độ tin cậy $1-\alpha$ giữ nguyên với mọi $\mathbf{a}$, ta có thể lập khoảng $T^2$ cho bất kỳ tổ hợp tuyến tính nào sau khi quan sát dữ liệu.
    \item \textbf{Ý nghĩa hình học:} Miền elip mô tả đồng thời miền khả dĩ của $(\mu_i, \mu_k)$, tích hợp cả biến động biên ($s_{ii}, s_{kk}$) lẫn tương quan mẫu ($s_{ik}$).
  \end{itemize}
\end{frame}
% ----------------------------------------------------------------------------
\begin{frame}{Khoảng $T^2$ là hình chiếu của ellipsoid tin cậy}
Các khoảng $T^2$ đồng thời cho từng thành phần chính là hình chiếu, của ellipsoid tin cậy lên các trục thành phần.

\vspace{0.5em}
\textbf{Ví dụ 5.4} (số liệu bức xạ vi ba, $n=42,\,p=2$, $F_{2,40}(.05)=3{,}23$):

\vspace{0.3em}
{\small
\[
\bar x_1\pm\sqrt{\tfrac{2(41)}{40}(3{,}23)}\sqrt{\tfrac{.0144}{42}}
\;\Rightarrow\; (.516,\ .612)
\]
\[
\bar x_2\pm\sqrt{\tfrac{2(41)}{40}(3{,}23)}\sqrt{\tfrac{.0146}{42}}
\;\Rightarrow\; (.555,\ .651)
\]}

\vspace{0.3em}
Hai khoảng này khớp với các khoảng $T^2$ trong Hình 5.2 (khung tiếp theo),
chính là hình chiếu của elip tin cậy 95\% (Ví dụ 5.3) lên hai trục $\mu_1,\mu_2$.
\end{frame}

\subsection{Ví dụ 5.5 — dữ liệu bài kiểm tra đại học (CLEP)}

% ----------------------------------------------------------------------------
\begin{frame}{Ví dụ 5.5: Dữ Liệu Điểm Thi \& Thiết Lập Hệ Số Tới Hạn}
Điểm số của $n=87$ sinh viên, trong 3 bài kiểm tra ứng với 3 biến: $X_1=$ khoa học xã hội \& lịch sử, $X_2=$ ngôn ngữ,
$X_3=$ khoa học.Ta có các số liệu sau:

\[
\bar{\mathbf x}=\begin{bmatrix}526{,}29\\54{,}69\\25{,}13\end{bmatrix},
\qquad
\mathbf S=\begin{bmatrix}
5808{,}06 & 597{,}84 & 222{,}03\\
597{,}84 & 126{,}05 & 23{,}39\\
222{,}03 & 23{,}39 & 23{,}11
\end{bmatrix}
\]
Xét các khoảng tin cậy đồng thời $95\%$ cho $\mu_1, \mu_2, \mu_3$
Ta có:
\[
\frac{p(n-1)}{n-p}F_{p,n-p}(\alpha)=\frac{3(86)}{84}F_{3,84}(.05)=\frac{3(86)}{84}(2{,}7)=8{,}29
\]
\end{frame}

% ----------------------------------------------------------------------------
\begin{frame}{Ví dụ 5.5: các khoảng $T^2$ đồng thời cho $\mu_1,\mu_2,\mu_3$}
Áp dụng (5-24) với $c^2=8{,}29$:

\begin{align*}
526{,}29 - \sqrt{8{,}29}\sqrt{\tfrac{5808{,}06}{87}} \le \mu_1 \le 526{,}29 + \sqrt{8{,}29}\sqrt{\tfrac{5808{,}06}{87}}
&\;\Rightarrow\; 503{,}06\le \mu_1\le 550{,}12\\[1ex]
54{,}69 - \sqrt{8{,}29}\sqrt{\tfrac{126{,}05}{87}} \le \mu_2 \le 54{,}69 + \sqrt{8{,}29}\sqrt{\tfrac{126{,}05}{87}}
&\;\Rightarrow\; 51{,}22\le \mu_2\le 58{,}16\\[1ex]
25{,}13 - \sqrt{8{,}29}\sqrt{\tfrac{23{,}11}{87}} \le \mu_3 \le 25{,}13 + \sqrt{8{,}29}\sqrt{\tfrac{23{,}11}{87}}
&\;\Rightarrow\; 23{,}65\le \mu_3\le 26{,}61
\end{align*}

\begin{itemize}
  \item Dữ liệu không sai lệch nghiêm trọng khỏi tính chuẩn (trừ khả năng
        ngoại lệ ở điểm ngôn ngữ); $n$ đủ lớn để biện minh phương pháp (Mục 5.5).
  \item Ba khoảng trên có thể \textbf{rộng hơn mức cần thiết}, vì với cùng
        $1-\alpha$ ta có thể hỏi trực tiếp về các \emph{hiệu}.
\end{itemize}
\end{frame}

% ----------------------------------------------------------------------------
\begin{frame}{Ví dụ 5.5: Ước lượng chênh lệch $\mu_2 - \mu_3$ \& Elip cho cặp biến}

=3{,}12.
\]
\[

\quad\Rightarrow\quad
26{,}44\le \mu_2-\mu_3\le 32{,}68.
\]

\vspace{0.3em}
\textbf{Elip tin cậy cho $(\mu_2,\mu_3)$}, theo (5-26):

{\small
\[
0{,}849(54{,}69-\mu_2)^2+4{,}633(25{,}13-\mu_3)^2
\]
\[
-\,2(0{,}859)(54{,}69-\mu_2)(25{,}13-\mu_3)\;\le\; 8{,}29
\]}

Elip này — cùng hai elip cho $(\mu_1,\mu_2)$ và $(\mu_1,\mu_3)$ — được vẽ trong
Hình 5.3.
\end{frame}

\begin{frame}{Phương pháp so sánh riêng lẻ từng biến}
Bỏ qua cấu trúc hiệp phương sai, chỉ dùng $\mathbf a'=[0,\ldots,a_i,\ldots,0]$
với $a_i=1$ (gợi ý bởi (5-21)):

{\footnotesize
\begin{equation*}
\tag{5-27}
\begin{aligned}
\bar x_1-t_{n-1}(\alpha/2)\sqrt{\tfrac{s_{11}}{n}}&\le \mu_1\le
\bar x_1+t_{n-1}(\alpha/2)\sqrt{\tfrac{s_{11}}{n}}\\
&\ \vdots\\
\bar x_p-t_{n-1}(\alpha/2)\sqrt{\tfrac{s_{pp}}{n}}&\le \mu_p\le
\bar x_p+t_{n-1}(\alpha/2)\sqrt{\tfrac{s_{pp}}{n}}
\end{aligned}
\end{equation*}}

\begin{itemize}
  \item Trước khi lấy mẫu, khoảng thứ $i$ có xác suất $1-\alpha$ phủ $\mu_i$.
  \item Nhưng xác suất để \textbf{tất cả} các khoảng cùng chứa các $\mu_i$
        tương ứng \textbf{không phải là} $1-\alpha$ nói chung.
\end{itemize}
\end{frame}

% ----------------------------------------------------------------------------
\begin{frame}{Hiện tượng sụt giảm độ tin cậy tổng thể khi các biến độc lập}
Xét trường hợp đặc biệt: phân phối chuẩn đồng thời và
\[
\boldsymbol\Sigma=\operatorname{diag}(\sigma_{11},\sigma_{22},\ldots,\sigma_{pp}).
\]
Khi đó các biến độc lập $\Rightarrow$ áp dụng quy tắc nhân xác suất:
\[
P[\text{tất cả khoảng $t$ trong (5-27) đều chứa }\mu_i]
=(1-\alpha)(1-\alpha)\cdots(1-\alpha)=(1-\alpha)^p
\]

\begin{block}{Ví dụ số}
$1-\alpha=0{,}95,\ p=6$ $\Rightarrow$ $(0{,}95)^6\approx 0{,}74$.
\end{block}

Dù mỗi khoảng ``tin cậy 95\%'', khẳng định \textbf{chung} cho cả 6 biến chỉ còn
đúng khoảng 74\%.
\end{frame}

% ----------------------------------------------------------------------------
\begin{frame}{Sự gia tăng độ rộng của khoảng tin cậy đồng thời $T^2$}
Để giữ xác suất $1-\alpha$ cho \textbf{tất cả} phát biểu cùng lúc, khoảng cho
từng thành phần phải rộng hơn khoảng $t$ riêng lẻ — mức độ phụ thuộc $p$, $n$,
$1-\alpha$.

\begin{block}{Ví dụ với: $1-\alpha=0{,}95,\ n=15,\ p=4$}
\[
\sqrt{\frac{4(14)}{11}(3{,}36)}=4{,}14
\qquad\text{so với}\qquad
t_{14}(.025)=2{,}145
\]
\[
100\left(\frac{4{,}14-2{,}145}{2{,}145}\right)\approx 93\%\ \text{rộng hơn}
\]
\end{block}
\end{frame}

\footnotesize Độ rộng tương đối của khoảng $T^2$ so với khoảng $t$ \textbf{tăng}

\end{frame}

\begin{frame}{Đánh giá phương pháp $T^2$ \& động lực cho phương pháp Bonferroni}
\begin{itemize}
  \item Mức tin cậy của \textbf{tập} khoảng $T^2$ luôn là $0{,}95$ (với $n,p$
        cố định); còn mức tin cậy \textbf{tổng thể} của tập khoảng $t$ riêng
        lẻ có thể nhỏ hơn $0{,}95$ rất nhiều $\Rightarrow$ hai bên không cùng
        ``mặt bằng''.
  \item Khoảng $t$ riêng lẻ vẫn là thông tin tốt nhất cho \emph{một} trung
        bình, nếu đó là suy luận duy nhất; một số nhà nghiên cứu chỉ tính
        chúng khi kiểm định $T^2$ đã bác bỏ $H_0$.
\end{itemize}

\vspace{0.5em}
\textbf{Ngược lại,} nếu \emph{chỉ} quan tâm $p$ trung bình thành phần, khoảng
$T^2$ lại \textbf{quá rộng}:
\begin{itemize}
  \item Nếu $\mu_1,\mu_2$ nằm trong khoảng $T^2$ của chúng thì $(\mu_1,\mu_2)$
        nằm trong hình chữ nhật tạo bởi hai khoảng đó — hình chữ nhật này
        \textbf{chứa elip tin cậy và còn hơn thế}.
  \item Elip (nhỏ hơn hình chữ nhật) đã có xác suất đúng $0{,}95$ phủ
        $\boldsymbol\mu$ $\Rightarrow$ xác suất phủ $(\mu_1,\mu_2)$ của hình
        chữ nhật \textbf{lớn hơn} $0{,}95$: độ tin cậy ``thừa'' không cần thiết.
\end{itemize}

$\Rightarrow$ Động lực cho cách tiếp cận thứ hai: \textbf{phương pháp Bonferroni}.
\end{frame}
