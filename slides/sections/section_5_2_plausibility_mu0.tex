\begin{frame}{Nhắc lại: bài toán một biến}
  \begin{block}{Câu hỏi}
    Giá trị cụ thể \(\mu_0\) có phải là một giá trị \textbf{hợp lý} cho trung bình tổng thể \(\mu\) hay không?
  \end{block}
  \vspace{-0.1cm}
  \[
    H_0: \mu = \mu_0
    \qquad\text{so với}\qquad
    H_1: \mu \ne \mu_0 \;\;\text{\small(đối hai phía)}
  \]
  \vspace{-0.35cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
    \small
    \textbf{Giả thiết:} \(X_1,\ldots,X_n\) là mẫu ngẫu nhiên từ một tổng thể \keyterm{chuẩn}.

    \vspace{0.15cm}
    \textbf{Thống kê kiểm định:}
    \[
      t = \frac{\bar{X} - \mu_0}{s/\sqrt{n}} \;\sim\; t_{n-1}
      \quad\text{khi } H_0 \text{ đúng.}
    \]
    \column{0.49\textwidth}
    \small
    \textbf{Trong đó:}
    \[
      \bar{X} = \frac{1}{n}\sum_{j=1}^{n} X_j,
      \qquad
      s^2 = \frac{1}{n-1}\sum_{j=1}^{n}(X_j - \bar{X})^2 .
    \]
    \vspace{-0.15cm}
    \begin{alertblock}{Bác bỏ \(H_0\) khi}
      \centering
      \(|t| > t_{n-1}(\alpha/2)\)
    \end{alertblock}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Nhìn \texorpdfstring{\(t^2\)}{t2} như một bình phương khoảng cách}
  \small
  Bác bỏ \(H_0\) khi \(|t|\) lớn \(\iff\) bác bỏ \(H_0\) khi \(t^2\) lớn:
  \begin{equation*}
    t^2
    = \frac{(\bar{X}-\mu_0)^2}{s^2/n}
    = \underbrace{n(\bar{X}-\mu_0)(s^2)^{-1}(\bar{X}-\mu_0)}_{\text{bình phương khoảng cách từ } \bar{X} \text{ đến } \mu_0}
    \tag{5-1}
  \end{equation*}
  \vspace{-0.2cm}
  \begin{center}

    \end{alertblock}

\end{frame}

\begin{frame}{Thống kê \texorpdfstring{\(T^2\)}{T2} của Hotelling}
  \vspace{-0.15cm}
  \begin{equation*}
    T^2
    = (\bar{\mathbf{X}}-\boldsymbol{\mu}_0)'
      \left(\frac{1}{n}\mathbf{S}\right)^{-1}
      (\bar{\mathbf{X}}-\boldsymbol{\mu}_0)
    = n(\bar{\mathbf{X}}-\boldsymbol{\mu}_0)'
      \mathbf{S}^{-1}
      (\bar{\mathbf{X}}-\boldsymbol{\mu}_0)
    \tag{5-4}
  \end{equation*}
  \vspace{-0.3cm}
  \small
  \[
    \bar{\mathbf{X}} = \frac{1}{n}\sum_{j=1}^{n}\mathbf{X}_j,
    \qquad
    \mathbf{S}
    = \frac{1}{n-1}\sum_{j=1}^{n}
      (\mathbf{X}_j-\bar{\mathbf{X}})(\mathbf{X}_j-\bar{\mathbf{X}})',
    \qquad
    \boldsymbol{\mu}_0 =
    \begin{bmatrix}
      \mu_{10} \\ \vdots \\ \mu_{p0}
    \end{bmatrix}
  \]
  \vspace{-0.1cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
    \begin{block}{Vì sao là \((1/n)\mathbf{S}\)?}
      Theo Kết quả 3.1: \(\operatorname{Cov}(\bar{\mathbf{X}})=\boldsymbol{\Sigma}/n\),
      nên \((1/n)\mathbf{S}\) là \textbf{ma trận hiệp phương sai ước lượng} của \(\bar{\mathbf{X}}\).
    \end{block}
    \column{0.47\textwidth}
    \begin{itemize}\setlength{\itemsep}{3pt}
      \item Đặt theo tên \keyterm{Harold Hotelling}, người đầu tiên tìm ra phân phối mẫu của thống kê này.
      \item \(T^2\) quá lớn \(\Rightarrow\) \(\bar{\mathbf{x}}\) ở quá xa \(\boldsymbol{\mu}_0\) \(\Rightarrow\) bác bỏ \(H_0\).
    \end{itemize}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Phân phối của \texorpdfstring{\(T^2\)}{T2} và quy tắc bác bỏ}
  \vspace{-0.1cm}
  \begin{block}{Kết quả}
    Cho \(\mathbf{X}_1,\ldots,\mathbf{X}_n\) là mẫu ngẫu nhiên từ \(N_p(\boldsymbol{\mu},\boldsymbol{\Sigma})\). Khi đó,
    \textbf{bất kể} giá trị thực của \(\boldsymbol{\mu}\) và \(\boldsymbol{\Sigma}\):
    \vspace{-0.2cm}
    \begin{gather*}
      T^2 \;\text{có phân phối như}\; \frac{(n-1)p}{n-p}\,F_{p,\,n-p}
      \tag{5-5}\\
      \alpha
      = P\!\left[
        n(\bar{\mathbf{X}}-\boldsymbol{\mu})'\mathbf{S}^{-1}(\bar{\mathbf{X}}-\boldsymbol{\mu})
        > \frac{(n-1)p}{n-p}F_{p,n-p}(\alpha)
      \right]
      \tag{5-6}
    \end{gather*}
    \vspace{-0.35cm}

    {\small \(F_{p,n-p}(\alpha)\): phân vị trên mức \(100\alpha\%\) của \(F_{p,n-p}\) \(\Rightarrow\) chỉ cần bảng \(F\), không cần bảng riêng cho \(T^2\).}
  \end{block}
  \vspace{-0.05cm}
  \begin{alertblock}{Kiểm định \(H_0:\boldsymbol{\mu}=\boldsymbol{\mu}_0\) so với \(H_1:\boldsymbol{\mu}\ne\boldsymbol{\mu}_0\) ở mức \(\alpha\): bác bỏ \(H_0\) nếu}
    \vspace{-0.2cm}
    \begin{equation*}
      T^2
      = n(\bar{\mathbf{x}}-\boldsymbol{\mu}_0)'\mathbf{S}^{-1}(\bar{\mathbf{x}}-\boldsymbol{\mu}_0)
      > \frac{(n-1)p}{n-p}F_{p,n-p}(\alpha)
      \tag{5-7}
    \end{equation*}
  \end{alertblock}
\end{frame}

\begin{frame}{Đối chiếu: một biến và đa biến}
  \vspace{0.1cm}
  \centering
  \small
  \renewcommand{\arraystretch}{1.55}
  \begin{tabular}{@{}p{2.7cm}p{4.9cm}p{5.9cm}@{}}
    \toprule
    & \textcolor{hustgray}{\textbf{Một biến}} & \textcolor{hustnavy}{\textbf{Đa biến}} \\

    ``Thước đo'' & \(s^2/n\): phương sai ước lượng của \(\bar{X}\) & \(\mathbf{S}/n\): hiệp phương sai ước lượng của \(\bar{\mathbf{X}}\) \\

  \end{tabular}

\end{frame}

\begin{frame}{Phân phối Wishart: phiên bản ma trận của \texorpdfstring{\(\chi^2\)}{chi2}}
  \begin{block}{Định nghĩa}
    Nếu \(\mathbf{Z}_1,\ldots,\mathbf{Z}_m\) độc lập, \(\mathbf{Z}_j\sim N_p(\mathbf{0},\boldsymbol{\Sigma})\), thì
    \(
      \mathbf{W}=\sum_{j=1}^{m}\mathbf{Z}_j\mathbf{Z}_j' \sim W_p(m,\boldsymbol{\Sigma})
    \)
    \;--- \(p\): số biến, \(m\): bậc tự do, \(\boldsymbol{\Sigma}\): ma trận hiệp phương sai tổng thể.
  \end{block}
  \vspace{0.05cm}
  \centering
  \begin{tikzpicture}[
      card/.style={line width=0.8pt, rounded corners=4pt,
        text width=5.7cm, minimum height=1.75cm, align=center, font=\small}
    ]
    \node[card, draw=hustgray, fill=hustgray!6] (u) {\textcolor{hustgray}{\textbf{Một biến}}\\[3pt]
      \((n-1)s^2=\sum_{j}(X_j-\bar{X})^2 \sim \sigma^2\chi^2_{n-1}\)};
    \node[card, draw=hustnavy, fill=hustnavy!6, right=1.3cm of u] (m) {\textcolor{hustnavy}{\textbf{Đa biến}}\\[3pt]
      \((n-1)\mathbf{S} \sim W_p(n-1,\boldsymbol{\Sigma})\)};
    \draw[-{Stealth[length=3mm]}, line width=1.3pt, hustred] (u) -- node[above, font=\scriptsize, text=hustred] {\(s^2\to\mathbf{S}\)} (m);
  \end{tikzpicture}

  \vspace{0.25cm}
  \raggedright
  \small
  \begin{itemize}
    \item Khi \(p=1\): \(\sum \mathbf{Z}_j\mathbf{Z}_j'\) trở thành \(\sum Z_j^2\) --- đúng dạng khi bình phương quen thuộc.
    \item Wishart mô tả sự dao động ngẫu nhiên của \keyterm{ma trận hiệp phương sai mẫu}.
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Cấu trúc của \texorpdfstring{\(T^2\)}{T2}: chuẩn -- Wishart -- chuẩn}
  \vspace{-0.1cm}
  \small
  \[
    T^2
    = \bigl(\sqrt{n}(\bar{\mathbf{X}}-\boldsymbol{\mu}_0)\bigr)'
    \Bigl[\tfrac{1}{n-1}\textstyle\sum_{j=1}^{n}(\mathbf{X}_j-\bar{\mathbf{X}})(\mathbf{X}_j-\bar{\mathbf{X}})'\Bigr]^{-1}
    \bigl(\sqrt{n}(\bar{\mathbf{X}}-\boldsymbol{\mu}_0)\bigr)
  \]
  \vspace{-0.15cm}
  \begin{equation*}
    T^2_{p,n-1}
    = \underbrace{N_p(\mathbf{0},\boldsymbol{\Sigma})'}_{\textcolor{hustnavy}{\text{vectơ chuẩn}}}
      \Bigl[\underbrace{\tfrac{1}{n-1}W_{p,n-1}(\boldsymbol{\Sigma})}_{\textcolor{hustred}{\text{Wishart / bậc tự do}}}\Bigr]^{-1}
      \underbrace{N_p(\mathbf{0},\boldsymbol{\Sigma})}_{\textcolor{hustnavy}{\text{vectơ chuẩn}}}
    \tag{5-8}
  \end{equation*}
  \vspace{-0.1cm}
  \[
    t^2_{n-1}
    = \underbrace{\bigl(\sqrt{n}(\bar{X}-\mu_0)\bigr)}_{\textcolor{hustnavy}{\text{chuẩn}}}
      \;\underbrace{(s^2)^{-1}}_{\textcolor{hustred}{(\chi^2/\text{bậc tự do})^{-1}}}\;
      \underbrace{\bigl(\sqrt{n}(\bar{X}-\mu_0)\bigr)}_{\textcolor{hustnavy}{\text{chuẩn}}}
  \]
  \vspace{-0.2cm}
  \begin{block}{Từ (5-8) đến (5-5)}
    Vectơ chuẩn và ma trận Wishart \textbf{độc lập} [xem (4-23)] \(\Rightarrow\) mật độ đồng thời là tích các mật độ biên
    \(\Rightarrow\) bằng giải tích suy ra phân phối \(\frac{(n-1)p}{n-p}F_{p,n-p}\) của \(T^2\).
  \end{block}
\end{frame}

\begin{frame}{Ví dụ 5.1: tính \texorpdfstring{\(T^2\)}{T2} --- trung bình và hiệp phương sai mẫu}

\begin{frame}{Ví dụ 5.1: tính \texorpdfstring{\(T^2\)}{T2} --- kết quả}

  \textbf{Bước 3 --- Ma trận \(\mathbf{S}\) và nghịch đảo}

    \mathbf{S}=
    \begin{bmatrix}
      4 & -3 \\ -3 & 9
    \end{bmatrix},

  \[
    T^2
    = 3\,[8-9,\ 6-5]\,
      \frac{1}{27}
      \begin{bmatrix} 9 & 3 \\ 3 & 4 \end{bmatrix}
      \begin{bmatrix} 8-9 \\ 6-5 \end{bmatrix}

      \begin{bmatrix} -1 \\ 1 \end{bmatrix}

\begin{frame}{Tính bất biến của \texorpdfstring{\(T^2\)}{T2} khi đổi đơn vị đo}
  \vspace{-0.1cm}
  \begin{block}{Phát biểu}
    \(T^2\) \textbf{không thay đổi} dưới các phép biến đổi
    \vspace{-0.15cm}
    \begin{equation*}
      \underset{(p\times1)}{\mathbf{Y}}
      = \underset{(p\times p)}{\mathbf{C}}\;
        \underset{(p\times1)}{\mathbf{X}}
      + \underset{(p\times1)}{\mathbf{d}},
      \qquad \mathbf{C} \text{ không suy biến.}
      \tag{5-9}
    \end{equation*}
  \end{block}
  \vspace{-0.05cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
    \small
    \textbf{Ví dụ:} định tâm rồi co giãn,
    \(X_i \mapsto a_i(X_i-b_i)\) với \(a_i>0\), như đổi nhiệt độ từ \textbf{Fahrenheit} sang \textbf{Celsius}:
    \[
      C = \tfrac{5}{9}\,(F - 32)
      \quad (a = \tfrac{5}{9},\ b = 32).
    \]
    Nhân trước tiếp bởi ma trận không suy biến bất kỳ vẫn cho dạng (5-9).

    \column{0.49\textwidth}
    \small
    \textbf{Các đại lượng sau biến đổi:}
    \begin{align*}
      \bar{\mathbf{y}} &= \mathbf{C}\bar{\mathbf{x}} + \mathbf{d}
        && \text{(Kết quả 3.6)}\\
      \mathbf{S}_y &= \mathbf{C}\mathbf{S}\mathbf{C}'
        && \text{(Kết quả 3.6)}\\
      \boldsymbol{\mu}_Y &= \mathbf{C}\boldsymbol{\mu}+\mathbf{d}
        && \text{(2-24), (2-45)}
    \end{align*}
    Giá trị giả thuyết tương ứng: \(\boldsymbol{\mu}_{Y,0}=\mathbf{C}\boldsymbol{\mu}_0+\mathbf{d}\).
  \end{columns}
\end{frame}

\begin{frame}{Chứng minh tính bất biến}
  \small
  \vspace{-0.2cm}
  \begin{align*}
    T^2_{\mathbf{y}}
    &= n(\bar{\mathbf{y}}-\boldsymbol{\mu}_{Y,0})'
       \mathbf{S}_y^{-1}
       (\bar{\mathbf{y}}-\boldsymbol{\mu}_{Y,0}) \\
    &= n\{\mathbf{C}(\bar{\mathbf{x}}-\boldsymbol{\mu}_0)\}'
       (\mathbf{C}\mathbf{S}\mathbf{C}')^{-1}
       \{\mathbf{C}(\bar{\mathbf{x}}-\boldsymbol{\mu}_0)\} \\
    &= n(\bar{\mathbf{x}}-\boldsymbol{\mu}_0)'
       \mathbf{C}'\,
       \textcolor{hustred}{(\mathbf{C}')^{-1}\mathbf{S}^{-1}\mathbf{C}^{-1}}\,
       \mathbf{C}
       (\bar{\mathbf{x}}-\boldsymbol{\mu}_0)
       && \textcolor{hustred}{\text{vì } \mathbf{C} \text{ khả nghịch}}\\
    &= n(\bar{\mathbf{x}}-\boldsymbol{\mu}_0)'
       \mathbf{S}^{-1}
       (\bar{\mathbf{x}}-\boldsymbol{\mu}_0)
     \;=\; T^2_{\mathbf{x}}
  \end{align*}
  \vspace{-0.1cm}
  \begin{alertblock}{Ý nghĩa}
    Kết luận của kiểm định \(T^2\) \textbf{không phụ thuộc} vào đơn vị đo hay gốc tọa độ được chọn cho các biến.
  \end{alertblock}
\end{frame}
